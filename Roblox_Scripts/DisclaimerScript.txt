-- ══════════════════════════════════════════
--  RESPONSIVE SIZE HELPER
--  Reads the actual viewport each time
--  so it works on any device/resolution.
-- ══════════════════════════════════════════
local camera = workspace.CurrentCamera

local function getResponsiveSize()
	local vp = camera.ViewportSize
	local sw, sh = vp.X, vp.Y

	-- Panel takes up 88% of screen width, capped between 300px and 680px.
	-- Height is locked to a 68:50 aspect ratio of the chosen width.
	local panelW = math.clamp(sw * 0.65, 220, 500)
	local panelH = math.clamp(panelW * (500 / 680), 220, 500)

	-- Scale all font sizes relative to panel width
	-- 680px panel = 100% scale, smaller panels shrink fonts proportionally
	local scale = panelW / 680

	return panelW, panelH, scale
end

-- ══════════════════════════════════════════
--  CLOSE FUNCTION
-- ══════════════════════════════════════════
local function closeUI()
	enableControls()

	local cW, cH = panel.AbsoluteSize.X, panel.AbsoluteSize.Y

	TweenService:Create(panel, TweenInfo.new(
		0.3,
		Enum.EasingStyle.Back,
		Enum.EasingDirection.In
		), {
			Size = UDim2.new(0, 0, 0, 0),
			BackgroundTransparency = 1,
		}):Play()

	TweenService:Create(overlay, TweenInfo.new(0.3), {
		BackgroundTransparency = 1,
	}):Play()

	task.delay(0.35, function()
		screenGui:Destroy()
	end)
end

ackBtn.MouseButton1Click:Connect(closeUI)
