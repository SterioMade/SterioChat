local function createSmoothTween(instance, properties, duration, easingStyle, easingDirection, callback)
	local tweenInfo = TweenInfo.new(
		duration or 0.3,
		easingStyle or Enum.EasingStyle.Quad,
		easingDirection or Enum.EasingDirection.Out
	)
	local tween = TweenService:Create(instance, tweenInfo, properties)
	if callback then
		tween.Completed:Connect(callback)
	end
	tween:Play()
	return tween
end

-- Disable default Roblox chat UI (robust, retried, and re-enforced)
--------------------------------------------------------------------
local function disableDefaultChatUI()
	pcall(function()
		StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.Chat, false)
	end)

	pcall(function()
		local inputCfg = TextChatService:WaitForChild("ChatInputBarConfiguration", 5)
		if inputCfg then
			inputCfg.Enabled = false
		end
	end)

	pcall(function()
		local bubbleCfg = TextChatService:WaitForChild("BubbleChatConfiguration", 5)
		if bubbleCfg then
			bubbleCfg.Enabled = false
		end
	end)

	pcall(function()
		local windowCfg = TextChatService:WaitForChild("ChatWindowConfiguration", 5)
		if windowCfg then
			windowCfg.Enabled = false
		end
	end)
end

local currentEmoteTrack = nil
local currentHumanoid = nil

local function stopEmote()
	if currentEmoteTrack then
		currentEmoteTrack:Stop()
		currentEmoteTrack = nil
	end
	if currentHumanoid then
		pcall(function() currentHumanoid:PlayEmote("") end)
		currentHumanoid = nil
	end
end

local function playEmote(emoteName)
	local character = localPlayer.Character
	if not character then return end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return end
	stopEmote()

	local success, result = pcall(function()
		currentHumanoid = humanoid
		return humanoid:PlayEmote(emoteName)
	end)
	if success and result then return end

	local animator = humanoid:FindFirstChildOfClass("Animator")
	if not animator then return end

	local EMOTE_ANIMATIONS = {
		["dance"]="rbxassetid://507771019",["wave"]="rbxassetid://507770239",

local function setupMovementDetection()
	local movementCheckConnection = nil
	local function onCharacterAdded(character)
		local humanoid = character:WaitForChild("Humanoid", 5)
		if not humanoid then return end
		if movementCheckConnection then movementCheckConnection:Disconnect() end
		movementCheckConnection = RunService.Heartbeat:Connect(function()
			if currentEmoteTrack or currentHumanoid then
				if humanoid.MoveDirection.Magnitude > 0.1 then stopEmote() end
			end
		end)
		character.AncestryChanged:Connect(function()
			if not character.Parent then
				if movementCheckConnection then
					movementCheckConnection:Disconnect()
					movementCheckConnection = nil
				end
				stopEmote()
			end
		end)
	end
	if localPlayer.Character then onCharacterAdded(localPlayer.Character) end
	localPlayer.CharacterAdded:Connect(onCharacterAdded)
	localPlayer.AncestryChanged:Connect(function()
		if not localPlayer.Parent then
			if movementCheckConnection then
				movementCheckConnection:Disconnect()
				movementCheckConnection = nil
			end
		end
	end)
end
setupMovementDetection()

local activeChatBubbles = {}

local function isWithinBubbleDistance(character)
	local camera = workspace.CurrentCamera
	if not camera or not character then return false end
	local root = character:FindFirstChild("HumanoidRootPart")
	if not root then return false end
	return (camera.CFrame.Position - root.Position).Magnitude <= CHAT_BUBBLE_DISTANCE
end

local function createChatBubble(player, message)
	-- Strip "Username: " prefix if present
	local colonPos = message:find(": ")
	if colonPos then
		message = message:sub(colonPos + 2)
	end
	if not player or not player.Character then return end
	if not isWithinBubbleDistance(player.Character) then return end
	local head = player.Character:FindFirstChild("Head")
	if not head then return end
	if message:match("^/e") then return end

	for _, child in ipairs(head:GetChildren()) do
		if child:IsA("BillboardGui") and child.Name == "StickerBubble" then
			child:Destroy()
		end
		if child:IsA("BillboardGui") and child.Name == "TypingHeadBubble" then
			child:Destroy()
		end
	end

	if player and player.Character then
		local h = player.Character:FindFirstChild("Head")
		if h then
			for _, child in ipairs(h:GetChildren()) do
				if child:IsA("BillboardGui") and child.Name == "TypingHeadBubble" then
					child:Destroy()
				end
			end
		end
	end

	local existing = activeChatBubbles[player]
	if existing and existing.label and existing.gui and existing.gui.Parent then
		existing.label.Text = message
		do
			local textService = game:GetService("TextService")
			local maxBubbleWidth = _G.MAX_CHAT_BUBBLE_WIDTH or 400
			local textSize = existing.label.TextSize
			local textBounds = textService:GetTextSize(message, textSize, existing.label.Font, Vector2.new(maxBubbleWidth, math.huge))
			local newWidth = math.clamp(textBounds.X + 16, 60, maxBubbleWidth)
			local newHeight = math.min(math.max(textBounds.Y + 12, 30), 120)
			local mainFrame = existing.gui:FindFirstChild("BubbleFrame")
			if mainFrame then
				mainFrame.Size = UDim2.new(0, newWidth, 0, newHeight)
				existing.gui.Size = UDim2.new(0, newWidth, 0, newHeight)
			end
		end
		if existing.expireTask then task.cancel(existing.expireTask) end
		local duration = math.min(math.max(#message * 0.12, 4), 30)
		existing.expireTask = task.delay(duration, function()
			if existing.gui and existing.gui.Parent then
				local mf = existing.gui:FindFirstChild("BubbleFrame")
				local tl = existing.gui:FindFirstChild("ChatText", true)
				local st = mf and mf:FindFirstChildOfClass("UIStroke")
				local fadeOut = TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
				if mf then TweenService:Create(mf, fadeOut, {BackgroundTransparency = 1}):Play() end
				if tl then TweenService:Create(tl, fadeOut, {TextTransparency = 1}):Play() end
				if st then TweenService:Create(st, fadeOut, {Transparency = 1}):Play() end
				task.delay(0.25, function()
					if existing.gui and existing.gui.Parent then existing.gui:Destroy() end
					if activeChatBubbles[player] == existing then activeChatBubbles[player] = nil end
				end)
			else
				if activeChatBubbles[player] == existing then activeChatBubbles[player] = nil end
			end
		end)
		return
	end

	-- Build new bubble
	do
		local playerHeight = 2.2
		if head then
			local headSizeY = head.Size.Y
			playerHeight = math.clamp(headSizeY / 2 + 1.5, 2.0, 3.6)
		end

		local billboard = Instance.new("BillboardGui")
		billboard.Name = "ChatBubble"
		billboard.Size = UDim2.new(0, 300, 0, 60)
		billboard.StudsOffset = Vector3.new(0, playerHeight, 0)
		billboard.AlwaysOnTop = true
		billboard.MaxDistance = 150
		billboard.Adornee = head
		billboard.Parent = head

		local uiScale = Instance.new("UIScale")
		uiScale.Scale = 1
		uiScale.Parent = billboard

		local connection
		connection = RunService.RenderStepped:Connect(function()
			if not player.Character then
				if connection then connection:Disconnect() end
				return
			end
			local withinDistance = isWithinBubbleDistance(player.Character)
			billboard.Enabled = withinDistance
			if withinDistance then
				local camera = workspace.CurrentCamera
				local root = player.Character:FindFirstChild("HumanoidRootPart")
				if camera and root then
					local dist = (camera.CFrame.Position - root.Position).Magnitude
					uiScale.Scale = math.clamp(16 / math.max(dist, 1), 0.7, 1)
				end
			end
		end)

		local textService = game:GetService("TextService")
		local maxBubbleWidth = _G.MAX_CHAT_BUBBLE_WIDTH or 400
		local textSize = 15
		local textBounds = textService:GetTextSize(message, textSize, Enum.Font.Gotham, Vector2.new(maxBubbleWidth, math.huge))
		local finalWidth = math.clamp(textBounds.X + 16, 60, maxBubbleWidth)
		local finalHeight = math.max(textBounds.Y + 12, 30)

		if textBounds.Y > 96 then
			textSize = 14
			textBounds = textService:GetTextSize(message, textSize, Enum.Font.Gotham, Vector2.new(maxBubbleWidth, math.huge))
			finalWidth = math.clamp(textBounds.X + 16, 60, maxBubbleWidth)
			finalHeight = math.max(textBounds.Y + 12, 30)
		end
		finalHeight = math.min(finalHeight, 120)

		billboard.Size = UDim2.new(0, finalWidth, 0, finalHeight)

		local mainFrame = Instance.new("Frame")
		mainFrame.Name = "BubbleFrame"
		mainFrame.BackgroundColor3 = Color3.fromRGB(30, 32, 38)
		mainFrame.BackgroundTransparency = 0.1
		mainFrame.BorderSizePixel = 0
		mainFrame.Size = UDim2.new(0, finalWidth, 0, finalHeight)
		mainFrame.Position = UDim2.new(0.5, 0, 0, 0)
		mainFrame.AnchorPoint = Vector2.new(0.5, 0)
		mainFrame.Parent = billboard

		local corner = Instance.new("UICorner")
		corner.CornerRadius = UDim.new(0, 10)
		corner.Parent = mainFrame

		local stroke = Instance.new("UIStroke")
		stroke.Thickness = 1.5
		stroke.Transparency = 0.2
		stroke.Color = Color3.fromRGB(255, 255, 255)
		stroke.Parent = mainFrame

		do
			local padding = Instance.new("UIPadding")
			padding.PaddingLeft = UDim.new(0, 4)
			padding.PaddingRight = UDim.new(0, 4)
			padding.PaddingTop = UDim.new(0, 6)
			padding.PaddingBottom = UDim.new(0, 6)
			padding.Parent = mainFrame
		end

		local textLabel = Instance.new("TextLabel")
		textLabel.Name = "ChatText"
		textLabel.BackgroundTransparency = 1
		textLabel.Font = Enum.Font.Gotham
		textLabel.TextSize = textSize
		textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
		textLabel.TextXAlignment = Enum.TextXAlignment.Center
		textLabel.TextYAlignment = Enum.TextYAlignment.Center
		textLabel.Position = UDim2.new(0, 0, 0, 0)
		textLabel.Size = UDim2.new(1, 0, 1, 0)
		textLabel.TextWrapped = true
		textLabel.Text = message
		textLabel.Parent = mainFrame

		activeChatBubbles[player] = { gui = billboard, label = textLabel, expireTask = nil }

		mainFrame.BackgroundTransparency = 1
		textLabel.TextTransparency = 1
		stroke.Transparency = 1
		mainFrame.Size = UDim2.new(0, finalWidth * 0.9, 0, finalHeight * 0.9)

		task.defer(function()
			local fadeIn = TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
			TweenService:Create(mainFrame, fadeIn, {BackgroundTransparency = 0.1, Size = UDim2.new(0, finalWidth, 0, finalHeight)}):Play()
			TweenService:Create(textLabel, fadeIn, {TextTransparency = 0}):Play()
			TweenService:Create(stroke, fadeIn, {Transparency = 0.2}):Play()
		end)

		local duration = math.min(math.max(#message * 0.12, 4), 30)
		local expireTask = task.delay(duration, function()
			if activeChatBubbles[player] and activeChatBubbles[player].gui == billboard then
				local fadeOut = TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
				TweenService:Create(mainFrame, fadeOut, {BackgroundTransparency = 1, Size = UDim2.new(0, finalWidth * 0.9, 0, finalHeight * 0.9)}):Play()
				TweenService:Create(textLabel, fadeOut, {TextTransparency = 1}):Play()
				TweenService:Create(stroke, fadeOut, {Transparency = 1}):Play()
				task.delay(0.25, function()
					if billboard and billboard.Parent then billboard:Destroy() end
					if activeChatBubbles[player] and activeChatBubbles[player].gui == billboard then
						activeChatBubbles[player] = nil
					end
				end)
			end
		end)

		activeChatBubbles[player].expireTask = expireTask

		local character = player.Character
		character.AncestryChanged:Connect(function()
			if not character.Parent and activeChatBubbles[player] and activeChatBubbles[player].gui == billboard then
				billboard:Destroy()
				activeChatBubbles[player] = nil
			end
		end)
	end
end

local function createStickerBubble(player, imageId)
	if not player or not player.Character then return end
	if not isWithinBubbleDistance(player.Character) then return end
	local head = player.Character:FindFirstChild("Head")
	if not head then return end

	if activeChatBubbles[player] and activeChatBubbles[player].gui then
		activeChatBubbles[player].gui:Destroy()
		activeChatBubbles[player] = nil
	end
	for _, child in ipairs(head:GetChildren()) do
		if child:IsA("BillboardGui") and child.Name == "StickerBubble" then
			child:Destroy()
		end
	end

	do
		local BUBBLE_SIZE = 64
		local IMAGE_SIZE = 44

		local billboard = Instance.new("BillboardGui")
		billboard.Name = "StickerBubble"
		billboard.Size = UDim2.new(0, BUBBLE_SIZE, 0, BUBBLE_SIZE)
		billboard.StudsOffset = Vector3.new(0, 1.6, 0)
		billboard.AlwaysOnTop = true
		billboard.MaxDistance = 120
		billboard.Adornee = head
		billboard.Parent = head

		local connection
		connection = RunService.RenderStepped:Connect(function()
			if not player.Character then
				if connection then connection:Disconnect() end
				return
			end
			billboard.Enabled = isWithinBubbleDistance(player.Character)
		end)

		local bubbleFrame = Instance.new("Frame")
		bubbleFrame.BackgroundColor3 = Color3.fromRGB(30, 32, 38)
		bubbleFrame.BackgroundTransparency = 0.1
		bubbleFrame.BorderSizePixel = 0
		bubbleFrame.Size = UDim2.new(1, 0, 1, 0)
		bubbleFrame.Parent = billboard

		local corner = Instance.new("UICorner")
		corner.CornerRadius = UDim.new(0, 16)
		corner.Parent = bubbleFrame

		local stroke = Instance.new("UIStroke")
		stroke.Thickness = 1.5
		stroke.Transparency = 0.2
		stroke.Color = Color3.fromRGB(255, 255, 255)
		stroke.Parent = bubbleFrame

		local image = Instance.new("ImageLabel")
		image.BackgroundTransparency = 1
		image.Image = imageId
		image.Size = UDim2.new(0, IMAGE_SIZE, 0, IMAGE_SIZE)
		image.Position = UDim2.new(0.5, 0, 0.5, 0)
		image.AnchorPoint = Vector2.new(0.5, 0.5)
		image.Parent = bubbleFrame

		bubbleFrame.Size = UDim2.new(0.85, 0, 0.85, 0)
		bubbleFrame.BackgroundTransparency = 1
		image.ImageTransparency = 1
		stroke.Transparency = 1

		task.defer(function()
			local ti = TweenInfo.new(0.18, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
			TweenService:Create(bubbleFrame, ti, {Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 0.1}):Play()
			TweenService:Create(image, ti, {ImageTransparency = 0}):Play()
			TweenService:Create(stroke, ti, {Transparency = 0.2}):Play()
		end)

		task.delay(4, function()
			if billboard and billboard.Parent then
				local fadeOut = TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
				TweenService:Create(bubbleFrame, fadeOut, {BackgroundTransparency = 1, Size = UDim2.new(0.85, 0, 0.85, 0)}):Play()
				TweenService:Create(image, fadeOut, {ImageTransparency = 1}):Play()
				TweenService:Create(stroke, fadeOut, {Transparency = 1}):Play()
				task.delay(0.2, function()
					if billboard and billboard.Parent then billboard:Destroy() end
				end)
			end
		end)
	end
end

Players.PlayerRemoving:Connect(function(plr)
	local data = activeChatBubbles[plr]
	if data then
		if data.expireTask then task.cancel(data.expireTask) end
		if data.gui and data.gui.Parent then data.gui:Destroy() end
		activeChatBubbles[plr] = nil
	end
	mutedPlayers[plr] = nil
	mutedByOthers[plr] = nil
	muteCooldowns[plr] = nil
end)

local thumbCache = {}
local function getThumbnail(userId)
	if thumbCache[userId] then return thumbCache[userId] end
	local ok, content = pcall(function()
		local url, _ = Players:GetUserThumbnailAsync(userId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size48x48)
		return url
	end)
	if ok and content then
		thumbCache[userId] = content
		return content
	end
	return ""
end

-- UPDATE SYSTEM MESSAGE COLORS (defined here so RGB loops can call it)
--------------------------------------------------------------------
local function updateSystemMessageColors()
	local cf = screenGui:FindFirstChild("ChatFrame")
	if not cf then return end
	local mf = cf:FindFirstChild("Messages")
	if not mf then return end
	for _, child in ipairs(mf:GetChildren()) do
		if child:IsA("Frame") then
			local usernameLabel = child:FindFirstChildWhichIsA("TextButton")
			if usernameLabel and usernameLabel.Text and usernameLabel.Text:find("%[SYSTEM%]") then
				child.BackgroundColor3 = systemMessageColor
				for _, descendant in ipairs(child:GetDescendants()) do
					if descendant:IsA("TextLabel") and descendant ~= usernameLabel then
						descendant.TextColor3 = systemTextColor
						break
					end
				end
				usernameLabel.TextColor3 = systemTextColor
			end
		end
	end
end

local function updateSystemMessageColors()
	local cf = screenGui:FindFirstChild("ChatFrame")
	if not cf then return end
	local mf = cf:FindFirstChild("Messages")
	if not mf then return end
	for _, child in ipairs(mf:GetChildren()) do
		if child:IsA("Frame") then
			local usernameLabel = child:FindFirstChildWhichIsA("TextButton")
			if usernameLabel and usernameLabel.Text and usernameLabel.Text:find("%[SYSTEM%]") then
				child.BackgroundColor3 = systemMessageColor
				for _, descendant in ipairs(child:GetDescendants()) do
					if descendant:IsA("TextLabel") and descendant ~= usernameLabel then
						descendant.TextColor3 = systemTextColor
						break
					end
				end
				usernameLabel.TextColor3 = systemTextColor
			end
		end
	end
end

local ChatSettingsRemote = ReplicatedStorage:WaitForChild("ChatSettingsRemote", 10)

saveSettings = function()
	if not ChatSettingsRemote then return end
	pcall(function()
		ChatSettingsRemote:InvokeServer("save", {
			transparency = CHAT_BG_TRANSPARENCY,
			bgColorR = chatBackgroundColor.R,
			bgColorG = chatBackgroundColor.G,
			bgColorB = chatBackgroundColor.B,
			rgbEnabled = rgbEnabled and true or false,
			rgbBgEnabled = rgbBackgroundEnabled and true or false,
			outlineThickness = _G.RGB_OUTLINE_THICKNESS,
			sysBgColorR = systemMessageColor.R,
			sysBgColorG = systemMessageColor.G,
			sysBgColorB = systemMessageColor.B,
			rgbSysBgEnabled = rgbSystemBgEnabled and true or false,
			sysTextColorR = systemTextColor.R,
			sysTextColorG = systemTextColor.G,
			sysTextColorB = systemTextColor.B,
			rgbSysTextEnabled = rgbSystemTextEnabled and true or false,
		})
	end)
end

local function loadSettings()
	if not ChatSettingsRemote then return end
	local ok, data = pcall(function() return ChatSettingsRemote:InvokeServer("load") end)
	if not ok or not data then return end
	if data.transparency then
		CHAT_BG_TRANSPARENCY = data.transparency
		chatFrame.BackgroundTransparency = CHAT_BG_TRANSPARENCY
		transparencySlider.Text = "Chat Transparency: " .. string.format("%.2f", CHAT_BG_TRANSPARENCY)
	end
	if data.bgColorR then
		chatBackgroundColor = Color3.new(data.bgColorR, data.bgColorG, data.bgColorB)
		chatFrame.BackgroundColor3 = chatBackgroundColor
		bgColorButton.BackgroundColor3 = chatBackgroundColor
	end
	if data.outlineThickness then
		_G.RGB_OUTLINE_THICKNESS = data.outlineThickness
		chatStroke.Thickness = _G.RGB_OUTLINE_THICKNESS
		if _G.THICKNESS_BUTTON then
			_G.THICKNESS_BUTTON.Text = "Outline Thickness: " .. tostring(_G.RGB_OUTLINE_THICKNESS)
		end
	end
	if data.rgbEnabled then
		task.delay(0.5, function()
			rgbEnabled = true
			_G.RGB_BUBBLES_ENABLED = true
			if rgbButton then rgbButton.Text = "Disable RGB Border" end
			if rgbConnection then rgbConnection:Disconnect(); rgbConnection = nil end
			local hue = 0
			rgbConnection = RunService.RenderStepped:Connect(function(dt)
				hue = (hue + dt * 0.4) % 1
				local color = Color3.fromHSV(hue, 1, 1)
				if chatStroke then
					chatStroke.Color = color
					chatStroke.Transparency = 0.3
					chatStroke.Thickness = _G.RGB_OUTLINE_THICKNESS
				end
			end)
		end)
	end
	if data.rgbBgEnabled then
		task.delay(0.5, function()
			rgbBackgroundEnabled = true
			if bgRGBButton then bgRGBButton.Text = "✨ RGB"; bgRGBButton.TextColor3 = Color3.fromRGB(100, 255, 200) end
			if rgbBackgroundConnection then rgbBackgroundConnection:Disconnect(); rgbBackgroundConnection = nil end
			local hue = 0
			rgbBackgroundConnection = RunService.RenderStepped:Connect(function(dt)
				hue = (hue + dt * 0.3) % 1
				local color = Color3.fromHSV(hue, 0.8, 0.3)
				if chatFrame then chatFrame.BackgroundColor3 = color end
				if bgColorButton and bgColorButton.Parent then bgColorButton.BackgroundColor3 = color end
			end)
		end)
	end
	if data.sysBgColorR and not data.rgbSysBgEnabled then
		task.delay(0.5, function()
			systemMessageColor = Color3.new(data.sysBgColorR, data.sysBgColorG, data.sysBgColorB)
			if sysBgColorButton and sysBgColorButton.Parent then
				sysBgColorButton.BackgroundColor3 = systemMessageColor
			end
			updateSystemMessageColors()
		end)
	end
	if data.rgbSysBgEnabled then
		task.delay(0.5, function()
			rgbSystemBgEnabled = true
			if sysBgRGBButton then sysBgRGBButton.Text = "✨ RGB"; sysBgRGBButton.TextColor3 = Color3.fromRGB(100, 255, 200) end
			if rgbSystemBgConnection then rgbSystemBgConnection:Disconnect(); rgbSystemBgConnection = nil end
			local hue = 0
			rgbSystemBgConnection = RunService.RenderStepped:Connect(function(dt)
				hue = (hue + dt * 0.3) % 1
				systemMessageColor = Color3.fromHSV(hue, 0.6, 0.4)
				if sysBgColorButton and sysBgColorButton.Parent then
					sysBgColorButton.BackgroundColor3 = systemMessageColor
				end
				updateSystemMessageColors()
			end)
		end)
	end
	if data.sysTextColorR and not data.rgbSysTextEnabled then
		task.delay(0.5, function()
			systemTextColor = Color3.new(data.sysTextColorR, data.sysTextColorG, data.sysTextColorB)
			if sysTextColorButton and sysTextColorButton.Parent then
				sysTextColorButton.BackgroundColor3 = systemTextColor
				sysTextColorButton.TextColor3 = Color3.fromRGB(0, 0, 0)
			end
			updateSystemMessageColors()
		end)
	end
	if data.rgbSysTextEnabled then
		task.delay(0.5, function()
			rgbSystemTextEnabled = true
			if sysTextRGBButton then sysTextRGBButton.Text = "✨ RGB"; sysTextRGBButton.TextColor3 = Color3.fromRGB(100, 255, 200) end
			if rgbSystemTextConnection then rgbSystemTextConnection:Disconnect(); rgbSystemTextConnection = nil end
			local hue = 0
			rgbSystemTextConnection = RunService.RenderStepped:Connect(function(dt)
				hue = (hue + dt * 0.3) % 1
				systemTextColor = Color3.fromHSV(hue, 0.7, 1)
				if sysTextColorButton and sysTextColorButton.Parent then
					sysTextColorButton.BackgroundColor3 = systemTextColor
				end
				updateSystemMessageColors()
			end)
		end)
	end
end

	emojiPickerFrame.Position = UDim2.new(0, xPosition, 0, yPosition)

local function richEscape(text)
	text = tostring(text or "")
	text = text:gsub("&", "&amp;")

	text = text:gsub(">", "&gt;")

	local c = role and role.color or {}
	local r = math.clamp(tonumber(c.r) or 255, 0, 255)
	local g = math.clamp(tonumber(c.g) or 255, 0, 255)
	local b = math.clamp(tonumber(c.b) or 255, 0, 255)

	return string.format("#%02X%02X%02X", r, g, b)
end

-- Animated RGB tag helpers
function rainbowTagHex()
	local color = Color3.fromHSV((os.clock() * 0.35) % 1, 0.9, 1)

	return string.format(
		"#%02X%02X%02X",
		math.floor(color.R * 255),
		math.floor(color.G * 255),
		math.floor(color.B * 255)
	)
end

function makeRichRolePrefix(role, forcedHex)
	if type(role) ~= "table" or not role.tag or role.tag == "" then
		return ""
	end

	local hex

	if role.rgb == true then
		hex = forcedHex or rainbowTagHex()
	else
		hex = roleColorToHex(role)
	end

	return '<font color="' .. hex .. '">[' .. richEscape(role.tag) .. ']</font> '
end

task.spawn(function()
	while task.wait(0.08) do
		if messagesFrame then
			local hex = rainbowTagHex()

			for _, obj in ipairs(messagesFrame:GetDescendants()) do
				if obj:IsA("TextButton") then
					local tag = obj:GetAttribute("RainbowTagText")
					local username = obj:GetAttribute("RainbowUsernameText")

					if tag and tag ~= "" and username and username ~= "" then
						obj.Text =
							'<font color="' .. hex .. '">[' .. richEscape(tag) .. ']</font> '
							.. richEscape(username)
					end
				end
			end
		end
	end
end)

local function refreshOwnerRoleList()
	if not roleList then return end

	for _, child in ipairs(roleList:GetChildren()) do
		if child:IsA("TextLabel") then
			child:Destroy()
		end
	end

	local count = 0

	for userId, role in pairs(ownerRoleCache) do
		if type(role) == "table" and role.tag then
			count += 1

			local label = Instance.new("TextLabel")
			label.BackgroundTransparency = 1
			label.RichText = true
			label.Text = tostring(userId) .. " → " .. makeRichRolePrefix(role)
			label.TextColor3 = Color3.fromRGB(230, 230, 235)
			label.Font = Enum.Font.Gotham
			label.TextSize = 12
			label.TextXAlignment = Enum.TextXAlignment.Left
			label.Size = UDim2.new(1, -8, 0, 20)
			label.ZIndex = 82
			label.Parent = roleList
		end
	end

	if count == 0 then
		local label = Instance.new("TextLabel")
		label.BackgroundTransparency = 1
		label.Text = "No custom tags yet."
		label.TextColor3 = Color3.fromRGB(160, 160, 170)
		label.Font = Enum.Font.Gotham
		label.TextSize = 12
		label.TextXAlignment = Enum.TextXAlignment.Left
		label.Size = UDim2.new(1, -8, 0, 20)
		label.ZIndex = 82
		label.Parent = roleList
	end
end

addBubble = function(text, textColor, userId, isSystem, roleInfo)
	local bubble = Instance.new("Frame")
	bubble:SetAttribute("IsSystemMessage", isSystem == true)
	bubble.BackgroundColor3 = isSystem and systemMessageColor or Color3.fromRGB(45, 47, 53)
	bubble.BackgroundTransparency = 1
	bubble.BorderSizePixel = 0
	bubble.Size = UDim2.new(1, 0, 0, 0)
	bubble.AutomaticSize = Enum.AutomaticSize.Y
	bubble.ZIndex = 12
	bubble.Parent = messagesFrame

	local hLayout = Instance.new("UIListLayout")
	hLayout.FillDirection = Enum.FillDirection.Horizontal
	hLayout.SortOrder = Enum.SortOrder.LayoutOrder
	hLayout.VerticalAlignment = Enum.VerticalAlignment.Top
	hLayout.Padding = UDim.new(0, 6)
	hLayout.Parent = bubble

	do
		local bubbleCorner = Instance.new("UICorner")
		bubbleCorner.CornerRadius = UDim.new(0, 12)
		bubbleCorner.Parent = bubble
		local padding = Instance.new("UIPadding")
		padding.PaddingLeft = UDim.new(0, 8); padding.PaddingRight = UDim.new(0, 8)
		padding.PaddingTop = UDim.new(0, 4); padding.PaddingBottom = UDim.new(0, 4)
		padding.Parent = bubble
	end

	local iconObj = nil
	if userId and userId ~= 0 then
		local thumb = getThumbnail(userId)
		if thumb ~= "" then
			local icon = Instance.new("ImageLabel")
			icon.BackgroundTransparency = 1
			icon.Image = thumb
			icon.Size = UDim2.new(0, 18, 0, 18)
			icon.LayoutOrder = 1
			icon.ZIndex = 13
			icon.Parent = bubble
			icon.ImageTransparency = 1
			local iconCorner = Instance.new("UICorner")
			iconCorner.CornerRadius = UDim.new(1, 0)
			iconCorner.Parent = icon
			iconObj = icon
		end
	end

	local usernameEnd = text:find(": ")
	local usernameText = text
	local messageText = ""
	if usernameEnd then
		usernameText = text:sub(1, usernameEnd + 1)
		messageText = text:sub(usernameEnd + 2)
	end

	local usernameLabel = Instance.new("TextButton")
	usernameLabel.RichText = true
	usernameLabel.AutoButtonColor = false
	usernameLabel.BackgroundTransparency = 1
	usernameLabel.Font = Enum.Font.Gotham
	usernameLabel.TextSize = 15
	usernameLabel.TextXAlignment = Enum.TextXAlignment.Left
	usernameLabel.TextWrapped = false
	local role = roleInfo

	if not role and userId and userId ~= 0 then
		role = ownerRoleCache[tostring(userId)]
	end

	if role and not isSystem then
		usernameLabel.Text = makeRichRolePrefix(role) .. richEscape(usernameText)

		if role.rgb == true then
			usernameLabel:SetAttribute("RainbowTagText", tostring(role.tag))
			usernameLabel:SetAttribute("RainbowUsernameText", tostring(usernameText))
		else
			usernameLabel:SetAttribute("RainbowTagText", "")
			usernameLabel:SetAttribute("RainbowUsernameText", "")
		end
	else
		usernameLabel.Text = richEscape(usernameText)
		usernameLabel:SetAttribute("RainbowTagText", "")
		usernameLabel:SetAttribute("RainbowUsernameText", "")
	end
	usernameLabel.TextColor3 = textColor
	usernameLabel.AutomaticSize = Enum.AutomaticSize.XY
	usernameLabel.Size = UDim2.new(0, 0, 0, 0)
	usernameLabel.LayoutOrder = 2
	usernameLabel.ZIndex = 13
	usernameLabel.Parent = bubble
	usernameLabel.TextTransparency = 1

	usernameLabel.MouseButton1Click:Connect(function()
		if not userId or userId == 0 then return end
		local target = Players:GetPlayerByUserId(userId)
		if not target then return end
		if target == Players.LocalPlayer then addSystem("You can't whisper to yourself"); return end
		if whisperTargetPlayer == target then
			whisperTargetPlayer = nil
			inputBox.PlaceholderText = "Type a message..."
			addSystem("Whisper canceled")
		else
			whisperTargetPlayer = target
			inputBox.PlaceholderText = "[Whisper to " .. target.Name .. "]"
			addSystem("Whispering to " .. target.Name)
		end
	end)

	usernameLabel.MouseButton2Click:Connect(function()
		if not userId or userId == 0 then return end
		local target = Players:GetPlayerByUserId(userId)
		if not target or target == Players.LocalPlayer then return end
		Players:ReportAbuse(target, "Inappropriate chat message", "")
	end)
	
	if messageText == "" then
		hLayout.FillDirection = Enum.FillDirection.Vertical
		usernameLabel.TextWrapped = true
		usernameLabel.AutomaticSize = Enum.AutomaticSize.Y
		usernameLabel.Size = UDim2.new(1, 0, 0, 0)
	end

	local messageLabel = nil
	if messageText ~= "" then
		messageLabel = Instance.new("TextLabel")
		messageLabel.BackgroundTransparency = 1
		messageLabel.Font = Enum.Font.Gotham
		messageLabel.TextSize = 15
		messageLabel.TextXAlignment = Enum.TextXAlignment.Left
		messageLabel.TextWrapped = true
		messageLabel.Text = messageText
		messageLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
		messageLabel.AutomaticSize = Enum.AutomaticSize.Y
		messageLabel.Size = UDim2.new(0, 200, 0, 0)
		messageLabel.LayoutOrder = 3
		messageLabel.ZIndex = 13
		messageLabel.Parent = bubble
		messageLabel.TextTransparency = 1

		task.defer(function()
			task.wait(0.1)
			local function updateMessageWidth()
				local bubbleWidth = bubble.AbsoluteSize.X
				if bubbleWidth > 0 then
					local iconWidth = (iconObj and (iconObj.AbsoluteSize.X + 6) or 0)
					local usernameWidth = usernameLabel.AbsoluteSize.X
					local availableWidth = math.max(bubbleWidth - iconWidth - usernameWidth - 22, 50)
					messageLabel.Size = UDim2.new(0, availableWidth, 0, 0)
				end
			end
			updateMessageWidth()
			usernameLabel:GetPropertyChangedSignal("AbsoluteSize"):Connect(updateMessageWidth)
			bubble:GetPropertyChangedSignal("AbsoluteSize"):Connect(updateMessageWidth)
		end)
	end

	task.defer(function()
		local fadeIn = TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
		TweenService:Create(bubble, fadeIn, {BackgroundTransparency = isSystem and 0.2 or 0.35}):Play()
		TweenService:Create(usernameLabel, fadeIn, {TextTransparency = 0}):Play()
		if messageLabel then TweenService:Create(messageLabel, fadeIn, {TextTransparency = 0}):Play() end
		if iconObj then TweenService:Create(iconObj, fadeIn, {ImageTransparency = 0}):Play() end
	end)

	task.defer(function()
		resizeCanvas()
		local contentHeight = listLayout.AbsoluteContentSize.Y
		if contentHeight > messagesFrame.AbsoluteSize.Y then
			messagesFrame.CanvasPosition = Vector2.new(0, contentHeight - messagesFrame.AbsoluteSize.Y + 8)
		end
	end)
end

local recentSystemMessages = {}

addSystem = function(text)
	text = tostring(text or "")
	text = text:gsub("^%s*%[SYSTEM%]%s*", "")

	local key = text:lower():gsub("%s+", " ")

	if recentSystemMessages[key] then
		return
	end

	recentSystemMessages[key] = true

	task.delay(5, function()
		recentSystemMessages[key] = nil
	end)

	addBubble("[SYSTEM] " .. text, systemTextColor, nil, true)
end

local sendMessage
sendMessage = function(msg)
	msg = string.gsub(msg, "^%s+", "")
	msg = string.gsub(msg, "%s+$", "")
	if string.lower(msg) == "/clear" then
		for _, child in ipairs(messagesFrame:GetChildren()) do
			if child:IsA("Frame") and child.Name ~= "UIListLayout" then child:Destroy() end
		end
		resizeCanvas()
		return
	end
	if msg == "" then return end
	if not ChatEvent or not ChatEvent:IsA("RemoteEvent") then
		addSystem("Error: Chat system not ready. Please rejoin.")
		return
	end
	local success, err = pcall(function()
		if whisperTargetPlayer then
			ChatEvent:FireServer(msg, whisperTargetPlayer)
		else
			ChatEvent:FireServer(msg, currentChatMode)
		end
	end)
	if not success then
		addSystem("Error sending message. Please try again.")
		warn("Failed to fire ChatEvent:", err)
	end
end

local typingDebounceThread = nil
local wasTyping = false
local lastTypingText = ""

local function sendTypingState(typing)
	if wasTyping == typing then return end
	wasTyping = typing
	pcall(function() TypingEvent:FireServer(typing, currentChatMode, whisperTargetPlayer) end)
	if typing then createHeadTypingBubble() else removeHeadTypingBubble() end
end

inputBox:GetPropertyChangedSignal("Text"):Connect(function()
	local text = inputBox.Text
	if typingDebounceThread then task.cancel(typingDebounceThread); typingDebounceThread = nil end
	if text ~= "" and text ~= lastTypingText then
		sendTypingState(true)
		lastTypingText = text
		typingDebounceThread = task.delay(3, function()
			sendTypingState(false); typingDebounceThread = nil
		end)
	elseif text == "" then
		sendTypingState(false); lastTypingText = ""
	end
end)

inputBox.FocusLost:Connect(function(enterPressed)
	sendTypingState(false)
	if typingDebounceThread then task.cancel(typingDebounceThread); typingDebounceThread = nil end
	lastTypingText = ""
	if enterPressed then
		local text = inputBox.Text
		inputBox.Text = ""
		sendMessage(text)
	end
end)

inputBox:GetPropertyChangedSignal("Text"):Connect(function()

local ContextActionService = game:GetService("ContextActionService")
local function OpenChatBySlash(actionName, inputState, inputObject)
	if inputState ~= Enum.UserInputState.Begin then return Enum.ContextActionResult.Pass end
	if inputBox and inputBox:IsFocused() then return Enum.ContextActionResult.Pass end
	if not chatFrame.Visible then setChatVisible(true); task.wait(0.05) end
	inputBox:CaptureFocus()
	if inputBox.Text:sub(1,1) == "/" or inputBox.Text:sub(1,1) == "-" then inputBox.Text = inputBox.Text:sub(2) end
	task.defer(function()
		if inputBox and inputBox.Parent then
			if inputBox.Text:sub(1,1) == "/" or inputBox.Text:sub(1,1) == "-" then inputBox.Text = inputBox.Text:sub(2); inputBox.CursorPosition = #inputBox.Text + 1 end
			inputBox.CursorPosition = #inputBox.Text + 1
		end
	end)
	task.delay(0.1, function()
		if inputBox and inputBox.Parent and inputBox:IsFocused() then
			if inputBox.Text:sub(1,1) == "/" or inputBox.Text:sub(1,1) == "-" then inputBox.Text = inputBox.Text:sub(2); inputBox.CursorPosition = #inputBox.Text + 1 end
		end
	end)
	return Enum.ContextActionResult.Sink
end
ContextActionService:BindActionAtPriority("OpenChatBySlash", OpenChatBySlash, false, Enum.ContextActionPriority.High.Value + 10, Enum.KeyCode.Slash)

refreshMutePlayerList = function()
	for _, child in ipairs(mutePlayerList:GetChildren()) do
		if child:IsA("Frame") and child.Name == "PlayerEntry" then child:Destroy() end
	end
	local oldNoResults = mutePlayerList:FindFirstChild("NoResultsLabel")
	if oldNoResults then oldNoResults:Destroy() end

	local searchTerm = string.lower(searchBox.Text or "")
	local players = {}
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= localPlayer then table.insert(players, player) end
	end
	table.sort(players, function(a, b) return a.Name:lower() < b.Name:lower() end)

	local visibleCount = 0
	for _, player in ipairs(players) do
		if searchTerm == "" or string.find(player.Name:lower(), searchTerm, 1, true) then
			createPlayerEntry(player)
			visibleCount = visibleCount + 1
		end
	end
	if visibleCount == 0 and #searchTerm > 0 then
		local noResults = Instance.new("TextLabel")
		noResults.Name="NoResultsLabel"; noResults.BackgroundTransparency=1
		noResults.Text="🔍 No players found"; noResults.Font=Enum.Font.Gotham
		noResults.TextSize=14; noResults.TextColor3=Color3.fromRGB(180,180,180)
		noResults.Size=UDim2.new(1,0,0,40); noResults.ZIndex=52; noResults.Parent=mutePlayerList
	end
end

		else
			mutedPlayers[targetPlayer]={player=targetPlayer, muteTime=os.time()}; muteCooldowns[targetPlayer]=os.time()+60
			actionButton.Text="Unmute"; actionButton.BackgroundColor3=Color3.fromRGB(70,90,110); baseColor=actionButton.BackgroundColor3

			do
				local mutedIcon = Instance.new("TextLabel")
				mutedIcon.BackgroundTransparency=1; mutedIcon.Text="🔇"; mutedIcon.Font=Enum.Font.GothamBold
				mutedIcon.TextSize=18; mutedIcon.TextColor3=Color3.fromRGB(255,100,100)

					task.delay(2, function()
						if mutedIcon and mutedIcon.Parent then createSmoothTween(mutedIcon, {TextTransparency=1}, 0.3) end
					end)
				end)
			end

			createSmoothTween(entry, {BackgroundColor3=Color3.fromRGB(80,50,50)}, 0.15, nil, nil, function() createSmoothTween(entry, {BackgroundColor3=Color3.fromRGB(35,37,43)}, 0.2) end)
			addSystem("🔇 Muted "..targetPlayer.Name); sendMuteNotification(targetPlayer, true)
		end

		local oldCL = entry:FindFirstChild("CooldownLabel")
		if oldCL then oldCL:Destroy() end
		local newCooldownLabel = Instance.new("TextLabel")
		newCooldownLabel.Name="CooldownLabel"; newCooldownLabel.BackgroundTransparency=1
		newCooldownLabel.Text="⏳60s"; newCooldownLabel.Font=Enum.Font.Gotham; newCooldownLabel.TextSize=11
		newCooldownLabel.TextColor3=Color3.fromRGB(180,180,180); newCooldownLabel.Size=UDim2.new(0,50,0,20)

		newCooldownLabel.ZIndex=53; newCooldownLabel.Parent=entry
		actionButton.BackgroundColor3=Color3.fromRGB(50,50,60); actionButton.Text="⏳"; actionButton.Active=false

			while muteCooldowns[targetPlayer] and os.time() < muteCooldowns[targetPlayer] do
				local rem = muteCooldowns[targetPlayer] - os.time()
				if rem > 0 and newCooldownLabel and newCooldownLabel.Parent then
					newCooldownLabel.Text = "⏳"..math.ceil(rem).."s"
				end
				task.wait(0.5)
			end
			if newCooldownLabel and newCooldownLabel.Parent then newCooldownLabel:Destroy() end
			if actionButton and actionButton.Parent then
				actionButton.BackgroundColor3=isPlayerMuted(targetPlayer) and Color3.fromRGB(70,90,110) or Color3.fromRGB(60,70,90)
				actionButton.Text=isPlayerMuted(targetPlayer) and "Unmute" or "Mute"
				actionButton.Active=true; baseColor=actionButton.BackgroundColor3
			end
			muteCooldowns[targetPlayer] = nil
		end)()
	end)
end

local originalSendMessage = sendMessage
sendMessage = function(msg)
	msg = string.gsub(msg, "^%s+", "")
	msg = string.gsub(msg, "%s+$", "")
	if msg == "" then return end
	if whisperTargetPlayer and isPlayerMuted(whisperTargetPlayer) then
		addSystem("🔇 " .. whisperTargetPlayer.Name .. " is muted"); return
	end
	originalSendMessage(msg)
end

local MuteEvent = ReplicatedStorage:WaitForChild("MuteEvent", 10)
if MuteEvent then
	MuteEvent.OnClientEvent:Connect(function(action, fromPlayer)
		if not fromPlayer or not fromPlayer:IsA("Player") then return end
		if action == "muted_by" then
			mutedByOthers[fromPlayer] = true
			addSystem("🔇 " .. fromPlayer.Name .. " has muted you.")
		elseif action == "unmuted_by" then
			mutedByOthers[fromPlayer] = nil
			addSystem("🔊 " .. fromPlayer.Name .. " unmuted you.")
		elseif action == "you_muted" then
			mutedPlayers[fromPlayer] = {player = fromPlayer, muteTime = os.time()}
			if fromPlayer and fromPlayer.Parent then
				addSystem("🔇 " .. fromPlayer.Name .. " is still muted.")
			end
		end
	end)
end
