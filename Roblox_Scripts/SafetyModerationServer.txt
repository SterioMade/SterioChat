local function normalizeForHardBlock(msg)
	msg = msg:lower()
	-- Handle special character substitutions BEFORE stripping
	msg = msg:gsub("|", "i")    -- sh|t → shit, b|tCH → bitch
	msg = msg:gsub("§", "s")    -- a§s → ass
	msg = msg:gsub("@", "a")    -- @ss → ass
	msg = msg:gsub("0", "o")
	msg = msg:gsub("1", "i")
	msg = msg:gsub("3", "e")
	msg = msg:gsub("4", "a")
	msg = msg:gsub("5", "s")
	msg = msg:gsub("7", "t")
	msg = msg:gsub("!", "i")
	msg = msg:gsub("%$", "s")
	-- REMOVE *ALL* NON-LETTERS
	msg = msg:gsub("[^%a]", " ")
	msg = msg:gsub("%s+", " ")
	return msg
end

local function collapseRuns(s)
	local out = {}
	local lastChar = nil
	for i = 1, #s do
		local ch = s:sub(i, i)
		if ch ~= lastChar then
			table.insert(out, ch)
			lastChar = ch
		end
	end
	return table.concat(out)
end

local function containsLetterSubstitutionInstruction(text)
	local lower = string.lower(text)
	local hasChangeVerb = lower:find("change", 1, true) or lower:find("turn", 1, true)
		or lower:find("swap", 1, true) or lower:find("replace", 1, true)
	local hasConnector = lower:find("into", 1, true) or lower:find(" to ", 1, true)
		or lower:find("for", 1, true) or lower:find("with", 1, true)
	if not (hasChangeVerb and hasConnector) then return false end
	for token in lower:gmatch("%a+") do
		if #token == 1 then return true end
	end
	return false
end

local function editDistance(a, b)
	local la, lb = #a, #b
	local d = {}
	for i = 0, la do d[i] = {[0] = i} end
	for j = 0, lb do d[0][j] = j end
	for i = 1, la do
		for j = 1, lb do
			local cost = (a:sub(i,i) == b:sub(j,j)) and 0 or 1
			d[i][j] = math.min(d[i-1][j]+1, d[i][j-1]+1, d[i-1][j-1]+cost)
		end
	end
	return d[la][lb]
end

local function isPlatformTypo(word)
	if #word < 4 or #word > 10 then return false, nil end
	for _, p in ipairs(PLATFORM_TYPO_WORDS) do
		if #p == #word then
			for i = 1, #word - 1 do
				local swapped = word:sub(1,i-1)..word:sub(i+1,i+1)..word:sub(i,i)..word:sub(i+2)
				if swapped == p then return true, p end
			end
		end
		if math.abs(#p - #word) <= 1 and editDistance(word, p) <= 1 then
			return true, p
		end
	end
	return false, nil
end

-- NEW: Detect Morse code / dot-dash spam
local function looksLikeMorseCode(text)
	local lower = text:lower()
	local tokens = {}
	for token in lower:gmatch("[%.%-]+") do
		table.insert(tokens, token)
	end
	local tokenCount = #tokens
	local morseCharCount = 0
	for _, token in ipairs(tokens) do
		morseCharCount += #token
	end
	local hasDot = lower:find("%.")
	local hasDash = lower:find("%-")
	if tokenCount >= 3 and hasDot and hasDash then
		return true
	end
	for _, token in ipairs(tokens) do
		if #token >= 5 then
			return true
		end
	end
	if tokenCount >= 2 and morseCharCount >= 6 and lower:match("[a-z]") then
		return true
	end
	return false
end

local function isGayVariant(word)
	local collapsed = collapseRuns(word)
	return collapsed == "gay" or collapsed == "gey"
end

	return false
end

local typingState = {}

TypingEvent.OnServerEvent:Connect(function(player, isTyping, mode, whisperTarget)
	if typingState[player] and typingState[player].thread then
		task.cancel(typingState[player].thread)
		typingState[player].thread = nil
	end

	local function getRecipients()
		local recipients = {}
		if mode == "whisper" and typeof(whisperTarget) == "Instance" and whisperTarget:IsA("Player") then
			table.insert(recipients, whisperTarget)
		elseif mode == "team" then
			local senderTeam = player.Team
			if senderTeam then
				for _, plr in ipairs(Players:GetPlayers()) do
					if plr ~= player and plr.Team == senderTeam then
						table.insert(recipients, plr)
					end
				end
			end
		else
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= player then
					table.insert(recipients, plr)
				end
			end
		end
		return recipients
	end

	if isTyping then
		typingState[player] = { mode = mode, target = whisperTarget }
		local recipients = getRecipients()
		for _, plr in ipairs(recipients) do
			TypingEvent:FireClient(plr, player, true)
		end
		typingState[player].thread = task.delay(5, function()
			for _, plr in ipairs(recipients) do
				pcall(function() TypingEvent:FireClient(plr, player, false) end)
			end
			typingState[player] = nil
		end)
	else
		local state = typingState[player]
		if state then
			local recipients = getRecipients()
			for _, plr in ipairs(recipients) do
				pcall(function() TypingEvent:FireClient(plr, player, false) end)
			end
			typingState[player] = nil
		end
	end
end)

MuteEvent.OnServerEvent:Connect(function(player, action, targetPlayer)
	if typeof(targetPlayer) ~= "Instance" or not targetPlayer:IsA("Player") then return end
	if targetPlayer == player then return end

	local muterId = player.UserId
	local targetId = targetPlayer.UserId

	if action == "mute" then
		if not serverMutes[muterId] then
			serverMutes[muterId] = {}
		end
		serverMutes[muterId][targetId] = true
		MuteEvent:FireClient(targetPlayer, "muted_by", player)
	elseif action == "unmute" then
		if serverMutes[muterId] then
			serverMutes[muterId][targetId] = nil
		end
		MuteEvent:FireClient(targetPlayer, "unmuted_by", player)
	end
end)

local function restoreMuteStateForPlayer(newPlayer)
	task.wait(2.5)
	for muterId, muteTable in pairs(serverMutes) do
		if muteTable[newPlayer.UserId] then
			local muterPlayer = Players:GetPlayerByUserId(muterId)
			if muterPlayer then
				MuteEvent:FireClient(newPlayer, "muted_by", muterPlayer)
			end
		end
	end
	if serverMutes[newPlayer.UserId] then
		for mutedId, _ in pairs(serverMutes[newPlayer.UserId]) do
			local mutedPlayer = Players:GetPlayerByUserId(mutedId)
			if mutedPlayer then
				MuteEvent:FireClient(newPlayer, "you_muted", mutedPlayer)
			end
		end
	end
end

local function notifyExistingPlayersOfRejoin(rejoinedPlayer)
	task.wait(2.5)
	for _, existingPlayer in ipairs(Players:GetPlayers()) do
		if existingPlayer == rejoinedPlayer then continue end
		if serverMutes[existingPlayer.UserId] and serverMutes[existingPlayer.UserId][rejoinedPlayer.UserId] then
			MuteEvent:FireClient(existingPlayer, "you_muted", rejoinedPlayer)
		end
		if serverMutes[rejoinedPlayer.UserId] and serverMutes[rejoinedPlayer.UserId][existingPlayer.UserId] then
			MuteEvent:FireClient(existingPlayer, "muted_by", rejoinedPlayer)
		end
	end
end

local function sanitizeTag(tag)
	tag = tostring(tag or ""):upper()
	tag = tag:gsub("[^A-Z0-9 _%-]", "")
	tag = tag:match("^%s*(.-)%s*$") or ""
	if #tag > 14 then
		tag = tag:sub(1, 14)
	end

end

local function clampColorTable(color)
	color = type(color) == "table" and color or {}

	return {
		r = math.clamp(tonumber(color.r) or 255, 0, 255),
		g = math.clamp(tonumber(color.g) or 255, 0, 255),
		b = math.clamp(tonumber(color.b) or 255, 0, 255),
	}
end

local function makeRole(tag, color)
	tag = sanitizeTag(tag)
	if tag == "" then return nil end

	color = type(color) == "table" and color or {}

	return {
		tag = tag,
		color = clampColorTable(color),
		rgb = color.rgb == true
	}
end

end

local function getTagForUserId(userId)
	local key = tostring(userId)

	local custom = ownerTagData.users[key]
	if custom and custom.tag then
		return custom
	end

	if ownerTagData.disabledDefaults and ownerTagData.disabledDefaults[key] then
		return nil
	end

	return getDefaultTagForUserId(userId)
end

local function resolveUserId(target)
	if typeof(target) == "Instance" and target:IsA("Player") then
		return target.UserId
	end

	if type(target) == "number" then
		return target
	end

	target = tostring(target or ""):match("^%s*(.-)%s*$") or ""

	local asNumber = tonumber(target)
	if asNumber then
		return asNumber
	end

	local plr = Players:FindFirstChild(target)
	if plr then
		return plr.UserId
	end

	local ok, id = pcall(function()
		return Players:GetUserIdFromNameAsync(target)
	end)

	if ok and id then
		return id
	end

	return nil
end

local function makeFuzzy(msg)
	local lower = string.lower(msg)
	-- Special character substitutions (bypass attempts like sh|t, a§s, b|tCH)
	lower = lower
		:gsub("|", "i")   -- sh|t, b|tch
		:gsub("§", "s")   -- a§s
		:gsub("%$", "s")   -- $hit
		:gsub("!", "i")   -- b!tch
		:gsub("¡", "i")
		:gsub("@", "a")   -- @ss
		:gsub("£", "e")
		:gsub("¥", "y")
		:gsub("€", "e")
		:gsub("ø", "o")
		:gsub("ð", "d")
		:gsub("þ", "th")
		:gsub("0", "o")
		:gsub("1", "i")
		:gsub("3", "e")
		:gsub("4", "a")
		:gsub("5", "s")
		:gsub("7", "t")
	local normalized = lower
		:gsub("à", "a"):gsub("á", "a"):gsub("â", "a"):gsub("ã", "a"):gsub("ä", "a"):gsub("å", "a")
		:gsub("è", "e"):gsub("é", "e"):gsub("ê", "e"):gsub("ë", "e")
		:gsub("ì", "i"):gsub("í", "i"):gsub("î", "i"):gsub("ï", "i")
		:gsub("ò", "o"):gsub("ó", "o"):gsub("ô", "o"):gsub("õ", "o"):gsub("ö", "o")
		:gsub("ù", "u"):gsub("ú", "u"):gsub("û", "u"):gsub("ü", "u")
		:gsub("ý", "y"):gsub("ÿ", "y")
		:gsub("ñ", "n"):gsub("ç", "c")
	local norm  = normalized:gsub("[^%w]", "")
	local fuzzy = collapseRuns(norm)
	return lower, norm, fuzzy
end

local function escapePattern(s)
	return s:gsub("[%(%)%.%+%-%*%?%[%]%^%$%%]", "%%%0")
end

local function containsAny(hay, list)
	for _, word in ipairs(list) do
		local e = escapePattern(word)
		if hay:match("%f[%a]" .. e .. "%f[%A]")
			or hay:match("^" .. e .. "%f[%A]")
			or hay:match("%f[%a]" .. e .. "$")
			or hay == word then
			return true, word
		end
	end
	return false, nil
end

local function containsFreeRobuxScam(lower, fuzzy)
	local joined = lower:gsub("[^a-z]", "")
	local hasFree =
		lower:match("%f[%a]free%f[%A]")
		and not FREE_SAFE_WORDS[lower]
		and not lower:match("%f[%a]free%a+")
	if not hasFree then
		return false
	end
	for _, word in ipairs(ROBUX_VARIANTS) do
		if lower:find(word, 1, true)
			or fuzzy:find(word, 1, true)
			or joined:find(word, 1, true) then
			return true
		end
	end
	if joined:find("bloxdrope", 1, true)
		or joined:find("bloxdrop", 1, true) then
		return true
	end
	return false
end

local function matchesWholePhrase(text, pattern)
	local escaped = pattern:gsub("[%(%)%.%+%-%*%?%[%]%^%$%%]", "%%%0")
	local pattern1 = "%f[%W]" .. escaped .. "%f[%W]"
	local pattern2 = "^" .. escaped .. "%f[%W]"
	local pattern3 = "%f[%W]" .. escaped .. "$"
	local pattern4 = "^" .. escaped .. "$"
	return text:match(pattern1) ~= nil or text:match(pattern2) ~= nil
		or text:match(pattern3) ~= nil or text:match(pattern4) ~= nil
end

local function containsSpacedStandaloneWord(lower, word)
	local parts = {}
	for i = 1, #word do
		table.insert(parts, word:sub(i, i))
	end
	local pattern = "%f[%a]" .. table.concat(parts, "[%s%p]*") .. "%f[%A]"
	return lower:find(pattern) ~= nil
end

local function isEnglishOnly(raw)
	for _, code in utf8.codes(raw) do
		if code >= 32 and code <= 126 then
			-- normal ASCII text, always fine
		else
			-- any non-ASCII character just needs to be on the whitelist,
			-- regardless of which emoji block it falls in
			local char = utf8.char(code)
			if not ALLOWED_EMOJIS[char] then
				return false
			end
		end
	end
	return true
end

local function hasExcessiveRepeats(text)
	local lower = string.lower(text)
	local trimmed = lower:match("^%s*(.-)%s*$") or lower
	if isCommonGreetingVariation(text, lower, trimmed) then
		return false
	end
	local maxRepeat = 0
	local currentChar = nil
	local currentCount = 0
	for i = 1, #text do
		local ch = text:sub(i, i)
		if ch == currentChar then
			currentCount += 1
		else
			if currentCount > maxRepeat then maxRepeat = currentCount end
			currentChar = ch
			currentCount = 1
		end
	end
	if currentCount > maxRepeat then maxRepeat = currentCount end
	return maxRepeat > 7
end

local function normalizePlatformCompact(raw)
	local s = tostring(raw or ""):lower()

	s = s
		:gsub("|", "i")
		:gsub("!", "i")
		:gsub("1", "i")
		:gsub("0", "o")
		:gsub("3", "e")
		:gsub("4", "a")
		:gsub("5", "s")
		:gsub("7", "t")
		:gsub("@", "a")

	s = s:gsub("[^a-z]", "")
	s = collapseRuns(s)

	return s
end

local function findPlatformAliasInCompact(compact, allowShort)
	compact = tostring(compact or "")

	for _, alias in ipairs(PLATFORM_ALIASES) do
		local cleanAlias = normalizePlatformCompact(alias)

		if #cleanAlias >= 4 and compact:find(cleanAlias, 1, true) then
			return true, cleanAlias
		end
	end

	if allowShort then
		for short, platform in pairs(SHORT_PLATFORM_ALIASES) do
			if compact == short then
				return true, platform
			end
		end
	end

	return false, nil
end

local function detectObfuscatedPlatform(raw)
	local lettersOnly = normalizePlatformCompact(raw)

	return false, nil
end

local function detectColorCodedApp(lower)
	-- Collapse repeated letters so "bluee", "blueee", "appppp" all normalize correctly
	local collapsed = collapseRuns(lower:gsub("[^a-z ]", " "))
	-- Also normalize "blew" → "blue" (common misspelling bypass: "blew up?" = "blue app?")
	local normalized = lower:gsub("blew", "blue"):gsub("bluw", "blue"):gsub("bloo", "blue")
	local appWords = {"app", "platform", "application", "thing", "chat", "message", "msg", "dm", "server", "link"}
	-- "up" added: "blew up?" = "blue app?" bypass
	for _, entry in ipairs(COLOR_APP_MAP) do
		-- Check original, collapsed, and normalized (blew→blue) for the color name
		if lower:find(entry.color, 1, true) or collapsed:find(entry.color, 1, true)
			or normalized:find(entry.color, 1, true) then
			for _, appWord in ipairs(appWords) do
				-- Check both original and collapsed/normalized for the app word
				if lower:find(appWord, 1, true) or collapsed:find(appWord, 1, true)
					or normalized:find(appWord, 1, true) then
					return true, entry.platform
				end
			end
		end
	end
	return false, nil
end

local function hasContactIntent(lower)
	local letters = lower:gsub("[^a-z]", "")
	local collapsed = collapseRuns(letters)

	-- ✅ ALWAYS ALLOW: "add me on roblox"
	if lower:match("%f[%a]add%s+me%s+on%s+roblox%f[%A]")
		or lower:match("%f[%a]add%s+me%s+on%s+roblox%s*$")
		or collapsed:find("addmeonroblox", 1, true) then
		return false
	end

	-- ✅ ALWAYS ALLOW: Talking ABOUT platform content (video, post, trend, etc.)
	-- "have you seen this YouTube video", "that TikTok about Goku", "the Instagram post"
	-- ANY mention of content words = conversational, not solicitation
	local contentWords = {
		"video", "vid", "clip", "meme", "post", "reel", "short", "shorts",
		"trend", "trending", "channel", "streamer", "stream", "live",
		"song", "music", "audio", "sound", "edit", "edits",

		"game", "gaming", "gameplay",
		"news", "article", "story",
		"playstation", "xbox", "nintendo", "console",
		"heard", "seen", "watched", "saw", "watch",
		"about", "like", "love", "hate",
	}
	for _, cw in ipairs(contentWords) do
		if lower:find("%f[%a]" .. cw .. "%f[%A]") then
			return false
		end
	end

	-- 🚫 BLOCK: Explicit solicitation phrases — unambiguously trying to redirect off-platform
	local hardSolicitPhrases = {
		"addmeon", "followmeon", "dmmeon", "contactmeon",
		"joinmyon", "joinmy",
		"heresmydiscord", "heresmysnap", "heresmyinsta",
		"mydiscord", "mysnap", "myinstagram", "mywhatsapp",

		"whatsyoursnap", "whatsyourinsta",
		"urdiscord", "ursnap", "urinsta", "urwhatsapp",
		"blewapp", "blewplatform",
	}
	for _, phrase in ipairs(hardSolicitPhrases) do
		if collapsed:find(phrase, 1, true) then
			return true
		end
	end

	-- 🚫 BLOCK: Direct "contact me" action verbs
	if lower:match("%f[%a]add%s+me%f[%A]")
		or lower:match("%f[%a]follow%s+me%f[%A]")
		or lower:match("%f[%a]dm%s+me%f[%A]")
		or lower:match("%f[%a]hmu%f[%A]")
		or lower:match("%f[%a]hit%s+me%s+up%f[%A]")
		or lower:match("%f[%a]contact%s+me%f[%A]")
		or lower:match("%f[%a]reach%s+me%f[%A]") then
		return true
	end

	-- 🚫 BLOCK: "meet me" / "meet up" — trying to arrange off-platform meetup
	if lower:match("%f[%a]meet%s+me%f[%A]") or lower:match("%f[%a]meet%s+up%f[%A]") then
		return true
	end

	-- 🚫 BLOCK: Asking for someone's specific account on a platform
	-- "whats your discord", "ur snap", "your instagram"
	-- Only block possessive + platform (not "have you seen Instagram")
	if collapsed:find("whatsyour") or collapsed:find("whatisyour") then
		local askPlatforms = {
			"discord", "snapchat", "instagram", "whatsapp",
			"telegram", "tiktok", "twitter", "twitch",
		}
		for _, p in ipairs(askPlatforms) do
			if collapsed:find(p, 1, true) then return true end
		end
	end
	if collapsed:find("ursnapchat") or collapsed:find("urdiscord")
		or collapsed:find("urinsta") or collapsed:find("urwhatsapp") then
		return true
	end

	-- 🚫 BLOCK: Short-form handle requests with possessive + ? 
	-- "ur dc?", "your ig?" — but NOT "is dc good?" or "whats on ig?"
	local hasPossessive = lower:match("%f[%a]ur%f[%A]") or lower:match("%f[%a]your%f[%A]")
	local isQuestion = lower:find("?") ~= nil
	if hasPossessive and isQuestion then
		local shortCodes = { "dc", "sc", "ig", "wa", "tt" }
		for _, code in ipairs(shortCodes) do
			if lower:match("%f[%a]" .. code .. "%f[%A]") then
				return true
			end
		end
	end

	-- 🚫 BLOCK: "join my X", "text me on X", "find me on X" + platform name
	local isRedirecting =
		lower:find("%f[%a]join%s+my%f[%A]") or
		lower:find("%f[%a]text%s+me%f[%A]") or
		lower:find("%f[%a]msg%s+me%f[%A]") or

		}
		for _, platform in ipairs(platformNames) do
			if collapsed:find(platform, 1, true) then return true end
		end
	end

	return false
end

local function isImVerbSentence(trimmed)
	if not (trimmed:match("^i[%s']*m%s+%a+")
		and not trimmed:match("^i[%s']*m%s+%a+%s*$")) then
		return false
	end
	-- CRITICAL: Do NOT allow early-return if sentence contains an EXTREME word
	-- "I'm a predator", "I'm a terrorist", "I'm a pedo" must NOT be bypassed
	local extremeCheck = trimmed:lower():gsub("[^a-z]", "")
	local extremeImWords = {
		"predator","predetor","terrorist","terorist","pedophile","pedofile",
		"paedophile","pedo","paedo","pred","rapist","murderer","killer",
	}
	for _, w in ipairs(extremeImWords) do
		if extremeCheck:find(w, 1, true) then
			return false  -- Do NOT early-return; let normal checks run
		end
	end
	return true
end

local function looksLikeLetterGibberish(trimmed, messageLength)

	local len = #trimmed
	if len < 4 or len > 20 then return false end
	if trimmed:match("[aeiouy]") then return false end
	return true
end

	}
	for _, p in ipairs(platforms) do
		if compact:match("^" .. p .. "/[%w%-_]+")
			or compact:match("[%s%p]" .. p .. "/[%w%-_]+")
			or compact:match(p .. "/[%w%-_]+") then
			return true
		end
	end
	return false
end

local function looksLikeImageUrl(s)
	local lower = string.lower(s)
	local imagePatterns = {
		"imgur%.com", "i%.imgur%.com", "cdn%.discordapp%.com", "discord%.gg", "discordapp%.com",
		"media%.discordapp%.com", "prntscr%.com", "prnt%.sc", "gyazo%.com", "puu%.sh",
		"tinypic%.com", "photobucket%.com", "flickr%.com", "instagram%.com", "i%.instagram%.com",
		"pbs%.twimg%.com", "pixiv%.net", "deviantart%.com", "artstation%.com", "redd%.it",
		"i%.redd%.it", "reddit%.com", "i%.redditmedia%.com", "cdn%.redditmedia%.com",
		"twimg%.com", "t%.co", "bit%.ly", "tinyurl%.com", "shortlink", "image%.", "img%.", "pic%.", "photo%.",
	}
	local imageExtensions = {
		"%.jpg", "%.jpeg", "%.png", "%.gif", "%.bmp", "%.webp", "%.svg", "%.ico", "%.tiff", "%.tif", "%.heic", "%.heif",
	}
	for _, pattern in ipairs(imagePatterns) do
		if lower:find(pattern) then return true end
	end
	for _, ext in ipairs(imageExtensions) do
		if lower:find(ext .. "[^%w]?") or lower:find(ext .. "$") then return true end
	end
	if lower:find("data:image/") then return true end
	return false
end

local function isPlatformContentConversation(text)
	local lower = tostring(text or ""):lower()

	local platformWords = {
		"tiktok", "tik tok", "youtube", "yt", "instagram", "insta",
		"snapchat", "snap", "discord", "twitch", "twitter", "facebook"
	}

	local contentWords = {
		"video", "vid", "clip", "meme", "post", "reel", "short", "shorts",
		"trend", "trending", "channel", "streamer", "stream", "live",
		"song", "music", "audio", "sound", "edit", "edits",
		"content", "creator", "youtuber", "tiktoker",
		"game", "gaming", "gameplay",
		"news", "article", "story",
		"heard", "seen", "watched", "saw", "watch",
		"about", "like", "love", "hate", "funny", "cool", "good", "bad"
	}

	local hasPlatform = false
	for _, p in ipairs(platformWords) do
		if lower:find(p, 1, true) then
			hasPlatform = true
			break
		end
	end

	if not hasPlatform then
		return false
	end

	for _, cw in ipairs(contentWords) do
		if lower:match("%f[%a]" .. cw .. "%f[%A]") then
			return true
		end
	end

	return false
end

local function checkSplitWords(player, currentMessage)
	if not messageHistory[player] or #messageHistory[player] < 1 then
		return false, nil
	end

	local recentMessages = {}
	local historySize = #messageHistory[player]

	-- FIX: look back further (was 2, now 5) so multi-message spelling bypasses
	-- ("s up" / "h is h" / "i was i" / "t for t") get caught
	for i = historySize, math.max(1, historySize - 5), -1 do
		table.insert(recentMessages, 1, messageHistory[player][i])
	end

	table.insert(recentMessages, currentMessage)
	
	-- NEW: Catch split phone-number sharing using number words
	-- Examples:
	-- "call" / "me" / "my num" / "is" / "five four three two"
	-- "my numbers" / "is" / "one two three four five six seven"
	do
		local NUMBER_WORDS = {
			["zero"] = "0", ["oh"] = "0", ["o"] = "0",
			["one"] = "1", ["won"] = "1",
			["two"] = "2", ["to"] = "2", ["too"] = "2",
			["three"] = "3", ["tree"] = "3",
			["four"] = "4", ["for"] = "4", ["fore"] = "4",
			["five"] = "5",
			["six"] = "6",
			["seven"] = "7",
			["eight"] = "8", ["ate"] = "8",
			["nine"] = "9",
		}

		local function normalizePhoneText(text)
			local s = tostring(text or ""):lower()

			-- Keep digits for phone detection, but remove separators like // . - _
			s = s
				:gsub("|", "i")
				:gsub("!", "i")
				:gsub("@", "a")
				:gsub("[^a-z0-9]", " ")
				:gsub("%s+", " ")

			return s:match("^%s*(.-)%s*$") or s
		end

		local function compactPhoneContext(text)
			local s = normalizePhoneText(text):gsub("%s+", "")

			-- Only for detecting words like c4llme / n0mber
			s = s
				:gsub("0", "o")
				:gsub("1", "i")
				:gsub("3", "e")
				:gsub("4", "a")
				:gsub("5", "s")
				:gsub("7", "t")

			return s
		end

		local function isCodedShotContext(text)
			local compact = compactPhoneContext(text)

			return compact:find("numbersaretheshots", 1, true)
				or compact:find("numbersareshots", 1, true)
				or compact:find("mynumbersaretheshots", 1, true)
				or compact:find("mynumbersareshots", 1, true)
				or compact:find("numberaretheshot", 1, true)
				or compact:find("numberareshot", 1, true)
				or compact:find("shotsarenumbers", 1, true)
				or compact:find("shotsarethenumbers", 1, true)
				or compact:find("shotisnumber", 1, true)
				or compact:find("shotmeansnumber", 1, true)
				or compact:find("shotmeansdigit", 1, true)
				or compact:find("shotsmeannumbers", 1, true)
				or compact:find("shotsmeandigits", 1, true)
		end

		local function hasNormalPhoneContext(text)
			local compact = compactPhoneContext(text)

			local hasCallOrText =
				compact:find("callme", 1, true)
				or compact:find("textme", 1, true)
				or compact:find("messageme", 1, true)

			local hasNumberIntro =
				compact:find("mynumber", 1, true)
				or compact:find("mynumbers", 1, true)
				or compact:find("mynum", 1, true)
				or compact:find("myphone", 1, true)
				or compact:find("numberis", 1, true)
				or compact:find("numbersis", 1, true)
				or compact:find("numis", 1, true)
				or compact:find("phoneis", 1, true)
				or compact:find("heresmynumber", 1, true)
				or compact:find("heresmyphone", 1, true)

			local asksForNumber =
				compact:find("givemeyournumber", 1, true)
				or compact:find("givemeurnumber", 1, true)
				or compact:find("whatsyournumber", 1, true)
				or compact:find("whatsurnumber", 1, true)
				or compact:find("watsyournumber", 1, true)
				or compact:find("watsurnumber", 1, true)

			return hasNumberIntro or asksForNumber or (hasCallOrText and hasNumberIntro)
		end

		local function hasPhoneSharingContext(text)
			return hasNormalPhoneContext(text) or isCodedShotContext(text)
		end

		local function extractPhoneDigitsFromText(text)
			local clean = normalizePhoneText(text)
			local digits = {}
			local currentRun = 0
			local longestRun = 0

			local function wordToDigit(word)
				word = tostring(word or ""):lower()
				word = word:gsub("[^a-z0-9]", "")

				if NUMBER_WORDS[word] then
					return NUMBER_WORDS[word]
				end

				-- NEW: coded "shot" phone bypass.
				-- Only extracts from patterns like:
				-- oneshot, two-shot, shotfour, three shot
				-- It does NOT block "four shot" alone unless phone context exists.
				for numberWord, digit in pairs(NUMBER_WORDS) do
					if word == numberWord .. "shot"
						or word == numberWord .. "shots"
						or word == "shot" .. numberWord
						or word == "shots" .. numberWord then
						return digit
					end
				end

				return nil
			end

			local previousTokenWasNumber = false
			local previousNumberDigit = nil

			for token in clean:gmatch("%S+") do
				local tokenDigits = ""
				local digit = wordToDigit(token)

				if digit then
					tokenDigits = digit
					previousTokenWasNumber = true
					previousNumberDigit = digit
				elseif token == "shot" or token == "shots" then
					-- "three shot" / "four shots"
					if previousTokenWasNumber and previousNumberDigit then
						-- Do not add another digit because the number word was already counted.
						-- This line exists only to keep the coded context connected.
						tokenDigits = ""
					end

					previousTokenWasNumber = false
					previousNumberDigit = nil
				elseif token:match("^%d+$") then
					tokenDigits = token
					previousTokenWasNumber = false
					previousNumberDigit = nil
				else
					previousTokenWasNumber = false
					previousNumberDigit = nil
				end

				if tokenDigits ~= "" then
					for i = 1, #tokenDigits do
						table.insert(digits, tokenDigits:sub(i, i))
					end

					currentRun = currentRun + #tokenDigits

					if currentRun > longestRun then
						longestRun = currentRun
					end
				elseif token == "shot" or token == "shots" then
					-- Keep the run alive for phrases like "one shot two shot three shot"
					currentRun = currentRun
				else
					currentRun = 0
				end
			end

			return table.concat(digits, ""), longestRun
		end
		
		local function isShotCodeMessage(text)
			local clean = normalizePhoneText(text)
			local tokens = {}

			for token in clean:gmatch("%S+") do
				table.insert(tokens, token)
			end

			for _, token in ipairs(tokens) do
				local word = tostring(token or ""):lower()
				word = word:gsub("[^a-z0-9]", "")

				for numberWord, _ in pairs(NUMBER_WORDS) do
					if word == numberWord .. "shot"
						or word == numberWord .. "shots"
						or word == "shot" .. numberWord
						or word == "shots" .. numberWord then
						return true
					end
				end
			end

			for i = 1, #tokens - 1 do
				local a = tokens[i]
				local b = tokens[i + 1]

				if NUMBER_WORDS[a] and (b == "shot" or b == "shots") then
					return true
				end

				if (a == "shot" or a == "shots") and NUMBER_WORDS[b] then
					return true
				end
			end

			return false
		end
		
		-- Check the CURRENT message only, using recent messages only as context.
		-- This fixes the bug where normal messages after a bypass kept getting blocked.
		do
			local currentOnly = recentMessages[#recentMessages] or currentMessage

			local previousParts = {}
			for i = math.max(1, #recentMessages - 8), #recentMessages - 1 do
				table.insert(previousParts, recentMessages[i])
			end

			local previousContext = table.concat(previousParts, " ")

			local currentDigits, currentLongestRun = extractPhoneDigitsFromText(currentOnly)
			local currentHasDigits = currentDigits ~= ""
			local currentIsShotCode = isShotCodeMessage(currentOnly)

			local previousHasNormalPhoneContext = hasNormalPhoneContext(previousContext)
			local currentHasNormalPhoneContext = hasNormalPhoneContext(currentOnly)

			local previousHasShotContext = isCodedShotContext(previousContext)

			-- CASE 1:
			-- "so my numbers are the shots" is allowed,
			-- but the next coded shot-number message is blocked.
			if previousHasShotContext and currentIsShotCode then
				return true, "pii_phone"
			end

			-- CASE 2:
			-- Normal phone sharing:
			-- "my num is" + "five four three two"
			-- "call me" + "5432"
			if currentHasDigits and (previousHasNormalPhoneContext or currentHasNormalPhoneContext) then
				if currentLongestRun >= 4 then
					return true, "pii_phone"
				end

				if #currentDigits >= 4 then
					return true, "pii_phone"
				end

				-- If the user already created phone context, even one digit-word after it is suspicious.
				if previousHasNormalPhoneContext then
					return true, "pii_phone"
				end
			end
		end
	end
	
	-- NEW: Catch split platform fragments across recent messages
	-- Examples:
	-- "dis" + "cord?" = discord
	-- "in" + "sta??" = insta
	-- "tik" + "tik??" = tiktik/tiktok bypass
	do
		local function cleanFragment(msg)
			return normalizePlatformCompact(msg)
		end

		local function isShortFragment(msg)
			local cleaned = cleanFragment(msg)
			return cleaned ~= "" and #cleaned <= 12
		end

		-- Check only suffixes ending at the CURRENT message.
		-- This prevents old bad messages from making a later normal message block.
		for startIndex = #recentMessages, math.max(1, #recentMessages - 3), -1 do
			local compact = ""
			local allShort = true

			for i = startIndex, #recentMessages do
				local frag = cleanFragment(recentMessages[i])

				if frag ~= "" then
					compact ..= frag
				end

				if not isShortFragment(recentMessages[i]) then
					allShort = false
					break
				end
			end

			if allShort and #compact >= 4 and #compact <= 24 then
				local hit, platform = findPlatformAliasInCompact(compact, false)
				if hit then
					local recentText = table.concat(recentMessages, " ")

					-- Do not block normal content talk like:
					-- "that tiktok edit was funny"
					-- "I saw a youtube video"
					if not isPlatformContentConversation(recentText) then
						return true, "contact_platform"
					end
				end
			end
		end
	end

	-- NEW: Catch acrostic / first-letter / last-letter bypasses
	-- Example style: "a is a" / "g are g" / "e is elephant" -> spells "age"
	do
		local firstLetterTokens = {}
		local lastTokenFirstLetterTokens = {}
		local lastTokenEndLetterTokens = {}
		local singleLetterTokens = {}
		local clueCount = 0

		local ACROSTIC_PII_KEYS = {
			["age"] = "pii_age",
			["old"] = "pii_age",
			["dob"] = "pii_age",
			["bday"] = "pii_age",
			["birthday"] = "pii_age",
			["birthdate"] = "pii_age",

			["realname"] = "pii_real_name",
			["fullname"] = "pii_real_name",
			["name"] = "pii_real_name",

			["phone"] = "pii_phone",
			["number"] = "pii_phone",
			["email"] = "pii_email",
			["gmail"] = "contact_platform",
			["address"] = "pii_address",
			["location"] = "pii_location",
		}

		local function isSingleLetter(token)
			return token and #token == 1 and token:match("^[a-z]$") ~= nil
		end

		local function normalizeAcrosticKey(text)
			text = tostring(text or ""):lower()
			text = text
				:gsub("@", "a")
				:gsub("4", "a")
				:gsub("0", "o")
				:gsub("1", "i")
				:gsub("3", "e")
				:gsub("5", "s")
				:gsub("7", "t")
			return text:gsub("[^a-z0-9]", "")
		end

		local function tokeniseMessage(msg)
			local clean = tostring(msg or ""):lower()
			clean = clean
				:gsub("@", "a")
				:gsub("4", "a")
				:gsub("0", "o")
				:gsub("1", "i")
				:gsub("3", "e")
				:gsub("5", "s")
				:gsub("7", "t")
				:gsub("[^a-z0-9]", " ")
				:gsub("%s+", " ")

			clean = clean:match("^%s*(.-)%s*$") or ""

			local tokens = {}
			for token in clean:gmatch("%S+") do
				table.insert(tokens, token)
			end

			return tokens
		end

		local function looksLikeLetterClue(tokens)
			if #tokens < 1 or #tokens > 5 then
				return false
			end

			local first = tokens[1]
			local last = tokens[#tokens]

			local firstIsLetter = isSingleLetter(first)
			local lastIsLetter = isSingleLetter(last)

			local hasConnector = false
			for _, token in ipairs(tokens) do
				if token == "is"
					or token == "are"
					or token == "for"
					or token == "means"
					or token == "equals"
					or token == "as" then
					hasConnector = true
					break
				end
			end

			if firstIsLetter then
				local sameAsLastStart = last:sub(1, 1) == first
				local sameAsLastEnd = last:sub(#last, #last) == first

				if hasConnector or sameAsLastStart or sameAsLastEnd or #tokens <= 2 then
					return true
				end
			end

			if lastIsLetter and #tokens <= 4 then
				return true
			end

			return false
		end

		for _, msg in ipairs(recentMessages) do
			local tokens = tokeniseMessage(msg)

			if #tokens >= 1 and #tokens <= 5 then
				local firstToken = tokens[1]
				local lastToken = tokens[#tokens]

				local firstLetter = firstToken:sub(1, 1)
				local lastTokenFirstLetter = lastToken:sub(1, 1)
				local lastTokenEndLetter = lastToken:sub(#lastToken, #lastToken)

				if firstLetter:match("^[a-z]$") then
					table.insert(firstLetterTokens, firstLetter)
				end

				if lastTokenFirstLetter:match("^[a-z]$") then
					table.insert(lastTokenFirstLetterTokens, lastTokenFirstLetter)
				end

				if lastTokenEndLetter:match("^[a-z]$") then
					table.insert(lastTokenEndLetterTokens, lastTokenEndLetter)
				end

				for _, token in ipairs(tokens) do
					if isSingleLetter(token) then
						table.insert(singleLetterTokens, token)
					end
				end

				if looksLikeLetterClue(tokens) then
					clueCount += 1
				end
			end
		end

		local function hasAcrosticContactContext()

			for _, msg in ipairs(recentMessages) do

			end

				or compact:find("whatisyour", 1, true)
				or compact:find("tellmeur", 1, true)
				or compact:find("tellmeyour", 1, true)
				or compact:find("givemeur", 1, true)
				or compact:find("givemeyour", 1, true)
				or compact:find("sendmeur", 1, true)
				or compact:find("sendmeyour", 1, true)
				or compact:find("addme", 1, true)
				or compact:find("addmeon", 1, true)
				or compact:find("dmme", 1, true)
				or compact:find("followme", 1, true)

		local function checkJoinedSequence(letters)
			if #letters < 2 then
				return false, nil
			end

			local joined = table.concat(letters, "")
			local joinedFuzzy = collapseRuns(joined)
			local contactContext = hasAcrosticContactContext()

			local function sequenceHas(word, allowShort)
				local cleanWord = normalizeAcrosticKey(word)

				if cleanWord == "" then
					return false
				end

				if #cleanWord < 3 and not allowShort then
					return false
				end

				if joined:find(cleanWord, 1, true) then
					return true
				end

				if joinedFuzzy:find(cleanWord, 1, true) then
					return true
				end

				return false
			end

			local function checkWordList(list, reason, allowShort)
				if not list then
					return false, nil
				end

				for k, v in pairs(list) do
					local word

					if type(k) == "number" then
						word = v
					else
						word = k
					end

					if type(word) == "string" and sequenceHas(word, allowShort) then
						return true, reason
					end
				end

				return false, nil
			end

			local function checkShortPlatformAliases()
				if not contactContext then
					return false, nil
				end

				if clueCount < 1 then
					return false, nil
				end

				local shortKeys = {
					["dc"] = true, -- Discord
					["yt"] = true, -- YouTube
					["fb"] = true, -- Facebook
					["ig"] = true, -- Instagram
					["tt"] = true, -- TikTok
					["sc"] = true, -- Snapchat
					["wa"] = true, -- WhatsApp
					["x"] = true,  -- X / Twitter
				}

				for key, _ in pairs(shortKeys) do
					if sequenceHas(key, true) then
						return true, "contact_platform"
					end
				end

				return false, nil
			end

			local hit, reason

			-- 1) Short contact aliases like dc / yt / ig / fb
			hit, reason = checkShortPlatformAliases()
			if hit then return true, reason end

			-- 2) Existing dangerous / blocked lists
			hit, reason = checkWordList(PROFANITY, "profanity", false)
			if hit then return true, reason end

			hit, reason = checkWordList(SELF_HARM, "self_harm", false)
			if hit then return true, reason end

			hit, reason = checkWordList(EXTREME, "extreme", false)
			if hit then return true, reason end

			hit, reason = checkWordList(HARASSMENT, "harassment", false)
			if hit then return true, reason end

			hit, reason = checkWordList(SEXUAL_ROOTS, "sexual_content", false)
			if hit then return true, reason end

			hit, reason = checkWordList(EXPLICIT_SLURS, "slur", false)
			if hit then return true, reason end

			hit, reason = checkWordList(HARD_SEXUAL_WORDS, "sexual_content", false)
			if hit then return true, reason end

			hit, reason = checkWordList(HARD_BLOCK_EXTREMIST_TERMS, "extremism", false)
			if hit then return true, reason end

			hit, reason = checkWordList(HARD_BLOCK_EXTREMIST_EXACT, "extremism", false)
			if hit then return true, reason end

			hit, reason = checkWordList(EXTREMIST_NAMES, "extremism", false)
			if hit then return true, reason end

			hit, reason = checkWordList(EXTREMIST_CALLS, "extremism", false)
			if hit then return true, reason end

			hit, reason = checkWordList(EXTREMIST_PEOPLE_STRICT, "extremism", false)
			if hit then return true, reason end

			hit, reason = checkWordList(EXTREMIST_ORGS_STRICT, "extremism", false)
			if hit then return true, reason end

			hit, reason = checkWordList(EXTREMIST_SLOGANS_STRICT, "extremism", false)
			if hit then return true, reason end

			hit, reason = checkWordList(SCAM_PATTERNS, "scam", false)
			if hit then return true, reason end

			hit, reason = checkWordList(ROBUX_VARIANTS, "scam", false)
			if hit then return true, reason end

			hit, reason = checkWordList(CONTACT_PHRASES, "contact_platform", false)
			if hit then return true, reason end

			-- 3) PII and off-platform acrostics need clue-looking messages
			if clueCount >= 2 then
				for word, piiReason in pairs(ACROSTIC_PII_KEYS) do
					if sequenceHas(word, false) then
						return true, piiReason
					end
				end

				hit, reason = checkWordList(OFF_PLATFORM_KEYS, "contact_platform", false)
				if hit then return true, reason end

				hit, reason = checkWordList(CONTACT_PLATFORMS, "contact_platform", false)
				if hit then return true, reason end

				hit, reason = checkWordList(FOREIGN_PROFANITY, "foreign_profanity", false)
				if hit then return true, reason end

				hit, reason = checkWordList(LATIN_FOREIGN_PROFANITY, "foreign_profanity", false)
				if hit then return true, reason end
			end

			return false, nil
		end

		local hit, reason = checkJoinedSequence(firstLetterTokens)
		if hit then return true, reason end

		hit, reason = checkJoinedSequence(lastTokenFirstLetterTokens)
		if hit then return true, reason end

local function containsSexualContent(lower, fuzzy)
	local joined = lower:gsub("[^a-z]", "")
	if BENIGN_SEX_SUBSTRINGS[joined] then
		return false
	end
	if joined:match("c+o+o+m+") then
		return true
	end
	for _, root in ipairs(SEXUAL_ROOTS) do
		local e = root:gsub("[%(%)%.%+%-%*%?%[%]%^%$%%]", "%%%0")
		if joined:match("%f[%a]" .. e .. "%f[%A]")
			or joined:match("^" .. e .. "$")
			or lower:match("%f[%a]" .. e .. "%f[%A]") then
			return true
		end
	end
	return false
end

local function scanTokensForBlockedContent(text)
	local lower = text:lower()

			or joinedFuzzy:find(w, 1, true) then
			return true, "sexual_content"
		end
	end

	for _, root in ipairs(SEXUAL_ROOTS) do
		local e = root:gsub("[%(%)%.%+%-%*%?%[%]%^%$%%]", "%%%0")
		if joinedLetters:match("%f[%a]" .. e .. "%f[%A]")
			or joinedLetters == root
			or joinedFuzzy:match("%f[%a]" .. e .. "%f[%A]") then
			return true, "sexual_content"
		end
	end

	-- FIX: Catch "ballcleaner" and similar nonsense sexual compounds
	if joinedLetters:find("ballcleaner", 1, true)
		or joinedLetters:find("ballwasher", 1, true)
		or joinedFuzzy:find("ballcleaner", 1, true) then
		return true, "sexual_content"
	end

	if lower:match("%f[%a]ss%s+rank%f[%A]")
		or lower:match("%f[%a]ss%s+skin%f[%A]")
		or lower:match("%f[%a]ss%s+car%s+skin%f[%A]")
		or lower:match("%f[%a]s%s+score%f[%A]") then
		return false, nil
	end

		local t = token:lower()
		local fuzzy = collapseRuns(t)

		if containsSexualContent(t, fuzzy) then
			return true, "sexual_content"
		end

		for _, bad in ipairs(PROFANITY) do
			if fuzzy == bad then
				return true, "profanity"
			end
		end
	end

	return false, nil
end

local function looksLikeKeyboardMash(text)
	local trimmed = text:match("^%s*(.-)%s*$") or text
	if trimmed:find("%s") then return false end
	local len = #trimmed
	if len < 7 then return false end
	local lower = trimmed:lower()
	local letters = lower:gsub("[^a-z]", "")
	local digits = trimmed:gsub("[^0-9]", "")
	-- Mixed letters + digits in one token = always spam
	if #letters >= 4 and #digits >= 2 and len >= 8 then
		return true
	end
	-- Pure letter mash checks
	if #letters >= 8 and #digits == 0 then
		local vowels = 0
		for i = 1, #letters do
			local ch = letters:sub(i, i)
			if ch=="a" or ch=="e" or ch=="i" or ch=="o" or ch=="u" then
				vowels = vowels + 1
			end
		end
		local ratio = vowels / #letters
		-- q not followed by u = keyboard mash
		if (lower:find("q[^u]") or lower:sub(-1) == "q") and ratio < 0.35 then
			return true
		end
		-- Extremely low vowel ratio in long strings
		if ratio < 0.15 and #letters >= 10 then
			return true
		end
		-- Multiple unusual letter pairs
		local unusualBigrams = {"qw","wq","qd","dq","qs","sq","qz","zq","xw","wx"}
		local unusualCount = 0
		for _, bg in ipairs(unusualBigrams) do
			local _, n = lower:gsub(bg, "")
			unusualCount = unusualCount + n
		end
		if unusualCount >= 2 then
			return true
		end
	end
	return false
end

	return blocked, reasons
end

local function checkSpam(player, msg)
	msg = msg:match("^%s*(.-)%s*$") or msg
	local userId = player.UserId
	local now = os.time()

	local s = spamState[userId]
	if not s then
		s = { lastText = "", count = 0, windowStart = now }
		spamState[userId] = s
	end

	if now - s.windowStart > SPAM_WINDOW_SECONDS then
		s.count = 0
		s.windowStart = now
	end

	if msg == s.lastText then
		s.count += 1
	else
		s.lastText = msg
		s.count = 1
	end

	return s.count < SPAM_MAX_IN_WINDOW, "spam_repeat"
end

local function containsBlockedAnatomyAnywhere(raw)
	raw = tostring(raw or "")
	raw = raw:sub(1, MAX_MESSAGE_LENGTH)

	-- FIX: Removed "isaac" and "ike" from hard anatomy block
	-- These are common names and cause massive false positives
	-- They are now only blocked in explicit slur/sexual context via other checks

	local lower = raw:lower()
	local letters = lower:gsub("[^a-z]", "")

	for word, _ in pairs(ANATOMY_STANDALONE) do
		if letters == word then
			return true
		end
		if letters:match("^(" .. word .. ")+$") then
			return true
		end
		if lower:match("%f[%a]" .. word .. "%f[%A]") then
			return true
		end
	end

	return false
end

local function containsAcrosticInstruction(msg)
	local lower = tostring(msg or ""):lower()
	lower = lower:gsub("[^a-z0-9%s]", " "):gsub("%s+", " ")

	local hasInstruction =
		lower:find("look at", 1, true)
		or lower:find("check", 1, true)
		or lower:find("read", 1, true)

	local hasLetterTrick =
		lower:find("first letter", 1, true)
		or lower:find("first letters", 1, true)
		or lower:find("last letter", 1, true)
		or lower:find("end letter", 1, true)
		or lower:find("middle letter", 1, true)

	local hasMessageWord =
		lower:find("message", 1, true)
		or lower:find("messages", 1, true)
		or lower:find("sentence", 1, true)
		or lower:find("question", 1, true)

	return hasInstruction and hasLetterTrick and hasMessageWord
end

local function shouldBlock(sender, rawMessage)
	-- 🔒 EARLIEST CHECK: Self-identification as dangerous person
	-- Catches: "I am a pred", "I'm a predator", "I am a terrorist", "I'm a pred (testing)"
	-- These were bypassing via isImVerbSentence early return
	do
		local m = collapseRuns(rawMessage:lower()
			:gsub("|", "i"):gsub("§", "s"):gsub("!", "i"):gsub("@", "a")
			:gsub("0", "o"):gsub("1", "i"):gsub("3", "e"):gsub("4", "a"):gsub("5", "s")
			:gsub("[^a-z]", ""))

		local dangerousIdentities = {
			-- predator variants
			"predator","predetor","predatorr","predd",
			-- terrorist variants
			"terrorist","terorist","terroristt",
			-- pedo variants
			"pedophile","pedofile","paedophile","paedo",
			"pdf","pdfile","pdff",   -- bypass attempts for "ped"
			-- rapist/murderer
			"rapist","murderer",
			-- child-specific
			"childpredator","childmolester","childabuser",
		}

		for _, word in ipairs(dangerousIdentities) do
			if m:find(word, 1, true) then
				return true, "extreme"
			end
		end

		-- "pred" is short so only block if combined with identity claim
		-- "im a pred", "i am a pred", "ima pred"
		if (m:find("imapred") or m:find("iamapred") or m:find("imapred") or m:find("iapred"))
			or (m:find("pred") and (m:find("ima") or m:find("iam") or m:find("iam"))) then
			return true, "extreme"
		end

		-- standalone "pred" or "pedo" as the entire message
		local trimmedRaw = rawMessage:match("^%s*(.-)%s*$") or rawMessage
		local trimmedLetters = collapseRuns(trimmedRaw:lower():gsub("[^a-z]", ""))
		if trimmedLetters == "pred" or trimmedLetters == "predd"
			or trimmedLetters == "pedo" or trimmedLetters == "paedo"
			or trimmedLetters == "pdf" or trimmedLetters == "pdff" then
			return true, "extreme"
		end
	end

	if looksLikeMorseCode(rawMessage) then
		return true, "encoded_message"
	end
	
	if containsLetterSubstitutionInstruction(rawMessage) then
		return true, "bypass_instruction"
	end

	do
		local trimmedMsg = rawMessage:match("^%s*(.-)%s*$") or rawMessage
		if #trimmedMsg <= 15 and not isPlatformContentConversation(rawMessage:lower()) then
			for token in trimmedMsg:gmatch("%a+") do
				if isPlatformTypo(token:lower()) then
					return true, "contact_platform"
				end
			end
			-- catches separator-obfuscated spellings like "sma/p/p/" or "s.n.a.p"
			local compact = collapseRuns(trimmedMsg:lower():gsub("[^a-z]", ""))
			if isPlatformTypo(compact) then
				return true, "contact_platform"
			end
		end
	end

	do
		local lower = rawMessage:lower()
		for word in lower:gmatch("%a+") do
			if SAFE_EXACT_WORDS[word] then
				return false, ""
			end
		end
	end

	if rawMessage:match("^rbxassetid://") or rawMessage:match("^rbxsatestid://") then
		return false, ""
	end

	local function normalizeForExtremism(text)
		local joined = text:lower()
			:gsub("[^a-z]", "")
		return collapseRuns(joined)
	end

	local function containsSSSturmbannfuhrer(norm)
		return norm:find("sssturmbannfuhrer", 1, true)
	end

	local function containsSturmbannfuhrer(norm)
		return norm:find("sturmbannfuhrer", 1, true)
	end

	local function containsSS(norm)
		local naziSSPatterns = {
			"waffenss", "schutzstaffel", "sssturmbannfuhrer",
			"sturmbannfuhrer", "ssreich", "ssdivision",
		}
		for _, p in ipairs(naziSSPatterns) do
			if norm:find(p, 1, true) then
				return true
			end
		end
		return false
	end

	do
		local norm = normalizeForExtremism(rawMessage)

		for _, term in ipairs(HARD_BLOCK_EXTREMIST_TERMS) do
			local t = term:gsub("[^a-z]", "")
			if norm:find(t, 1, true) then
				return true, "extremism"
			end
		end

		for _, term in ipairs(HARD_BLOCK_EXTREMIST_EXACT) do
			local t = term:gsub("[^a-z]", "")
			if norm:find(t, 1, true) then
				return true, "extremism"
			end
		end

		if containsSSSturmbannfuhrer(norm) then
			return true, "extremism"
		end

		for _, p in ipairs(EXTREMIST_PEOPLE_STRICT) do
			if norm:find(p:gsub("%s+", ""), 1, true) then
				return true, "extremism"
			end
		end

		for _, o in ipairs(EXTREMIST_ORGS_STRICT) do
			if norm:find(o:gsub("%s+", ""), 1, true) then
				return true, "extremism"
			end
		end

		if containsSturmbannfuhrer(norm) and containsSS(norm) then
			return true, "extremism"
		end

		for _, s in ipairs(EXTREMIST_SLOGANS_STRICT) do
			if norm:find(s:gsub("%s+", ""), 1, true) then
				return true, "extremism"
			end
		end
	end

	if containsBlockedAnatomyAnywhere(rawMessage) then
		return true, "sexual_content"
	end

	-- FIX: Detect heavily obfuscated platform names like "d,c....cc?" or "d.i.s.c.o.r.d"
	-- Only block if message is SHORT or has actual contact intent
	do
		local obfHit, obfPlatform = detectObfuscatedPlatform(rawMessage)
		if obfHit then
			local msgTrimmed = rawMessage:match("^%s*(.-)%s*$") or rawMessage
			local msgLen = #msgTrimmed
			local lower = rawMessage:lower()

			-- Allow normal platform-content talk:
			-- "that tiktok edit was funny"
			-- "I saw a YouTube video"
			-- "that Instagram post was cool"
			if not isPlatformContentConversation(lower) then
				if hasContactIntent(lower) then
					return true, "contact_platform"
				end

				-- Still block short platform-only messages:
				-- "discord?"
				-- "insta?"
				-- "tiktok?"
				-- "yt?"
				if msgLen <= 15 then
					return true, "contact_platform"
				end
			end
		end
	end

	-- FIX: extremism keyword-only check — now uses WHOLE WORD check to prevent
	-- "lass" → false "ss" hit, "language" → false "nga" hit, etc.
	do
		local lower = rawMessage:lower()
		-- Apply special char substitutions for bypass attempts
		local lowerNorm = lower
			:gsub("|", "i"):gsub("§", "s"):gsub("!", "i"):gsub("@", "a")
			:gsub("0", "o"):gsub("1", "i"):gsub("3", "e"):gsub("4", "a"):gsub("5", "s")

		-- Only block "nazi" if it appears as a standalone word
		if lowerNorm:match("%f[%a]nazi%f[%A]") or lowerNorm:match("%f[%a]nazis%f[%A]") then
			return true, "extremism"
		end

		-- Only block "hitler" / misspellings as standalone word
		if lowerNorm:match("%f[%a]hitler%f[%A]")
			or lowerNorm:match("%f[%a]hitier%f[%A]")
			or lowerNorm:match("%f[%a]bitler%f[%A]") then  -- "heil bitler"
			return true, "extremism"
		end

		-- swastika — was completely missing from all checks
		local joinedNorm = collapseRuns(lowerNorm:gsub("[^a-z]", ""))
		if joinedNorm:find("swastika", 1, true) or joinedNorm:find("svastika", 1, true) then
			return true, "extremism"
		end

		-- osama bin laden references
		if lowerNorm:match("%f[%a]osama%f[%A]") then
			return true, "extremism"
		end
		if joinedNorm:find("osamabinladen", 1, true) or joinedNorm:find("binladen", 1, true) then
			return true, "extremism"
		end

		-- eps island / little saint james (Epstein references from screenshot)
		if joinedNorm:find("epsisland", 1, true)
			or joinedNorm:find("epsteinisland", 1, true)
			or joinedNorm:find("littlesaintjames", 1, true)
			or (lowerNorm:find("eps", 1, true) and lowerNorm:find("island", 1, true))
			or (lowerNorm:find("little", 1, true) and lowerNorm:find("saint", 1, true) and lowerNorm:find("james", 1, true)) then
			return true, "extremism"
		end

		-- heil + partial word detection (catches "Heil Na**", "Heil Nati...")
		if lowerNorm:match("heil%s*na") or lowerNorm:match("heil%s*hit")
			or lowerNorm:match("heil%s*bit") or lowerNorm:match("sieg%s*hei") then
			return true, "extremism"
		end

		-- "jew" as standalone with hate context
		-- Only block if there's hate intent, not just mentioning the word
		local hasJew = lowerNorm:match("%f[%a]jew%f[%A]") or lowerNorm:match("%f[%a]jews%f[%A]")
		if hasJew then
			local hateVerbs = {"hate","kill","gas","hang","die","burn","shoot","exterminate","eradicate"}
			for _, v in ipairs(hateVerbs) do
				if lowerNorm:find(v, 1, true) then
					return true, "hate_violence"
				end
			end
			-- "shut up jew" — insult standalone
			if lowerNorm:match("shut%s*up%s*jew") or lowerNorm:match("stupid%s*jew")
				or lowerNorm:match("dirty%s*jew") or lowerNorm:match("f+%s*jew")
				or joinedNorm:find("shutupjew", 1, true) then
				return true, "hate_violence"
			end
		end

		-- "the blax suk" — racial slur + insult
		if joinedNorm:find("blax", 1, true) then
			return true, "profanity"
		end

		-- "i hate jews" / "i hate [group]"
		for _, group in ipairs({"jews","muslims","blacks","whites","gays","christians","asians"}) do
			if lowerNorm:find("i hate " .. group, 1, true)
				or lowerNorm:find("i h8 " .. group, 1, true)
				or joinedNorm:find("ihate" .. group, 1, true) then
				return true, "hate_violence"
			end
		end
	end

	do
		local joined = collapseRuns(rawMessage:lower():gsub("[^a-z]", ""))
		if joined:match("^(vagina)+$") then
			return true, "sexual_content"
		end
		if joined:match("^e+z+$") or joined:match("^z+z+z+$") then
			return true, "toxic_spam"
		end
	end

	do
		local lwr = rawMessage:lower()
		local fzy = collapseRuns(lwr:gsub("[^a-z]", ""))
		if containsLatinForeignProfanity(lwr, fzy) then
			return true, "profanity"
		end
	end

	local okFront, reasonFront = hardBlockPII(rawMessage)
	if not okFront then
		return true, reasonFront
	end

	do
		local normalized = rawMessage
		normalized = normalizeForSafePhrase(normalized)
		normalized = collapseRuns(normalized:gsub("[^a-z]", ""))
		for _, phrase in ipairs(EXTREMIST_CALLS) do
			if normalized:find(phrase, 1, true) then
				return true, "extremism"
			end
		end
	end

	local lowerForHate = rawMessage:lower()
	if isHateViolence(rawMessage)
		and not isNormalConversationalSentence(rawMessage, lowerForHate, lowerForHate:match("^%s*(.-)%s*$") or lowerForHate)
		and not containsSafePhrasePattern(rawMessage) then
		return true, "hate_violence"
	end

	if isExtremistPraise(rawMessage) then
		return true, "extremism"
	end

	local lower = rawMessage:lower():match("^%s*(.-)%s*$") or rawMessage:lower()
	for _, phrase in ipairs(SAFE_MULTIWORD_PHRASES) do
		if lower == phrase then
			return false, ""
		end
	end
	
	if containsAcrosticInstruction(rawMessage) then
		return true, "bypass_instruction"
	end

	local isSplit, splitReason = checkSplitWords(sender, rawMessage)
	if isSplit then
		return true, splitReason
	end

	local okSpam, spamReason = checkSpam(sender, rawMessage)
	if not okSpam then
		return true, spamReason
	end

	local blocked, reasons = analyseMessage(rawMessage)
	if blocked then
		if reasons then
			for _, r in ipairs(reasons) do
				if r == "scam" then
					return true, "scam"
				end
			end
			for _, r in ipairs(reasons) do
				if type(r) == "string" and r:find("pii_") then
					return true, r
				end
			end
		end
		return true, reasons[1] or "safety"
	end

	return false, ""
end

local function robloxFilter(fromPlayer, rawMessage, toPlayer)
	local ok, result = pcall(function()
		local filterResult = TextService:FilterStringAsync(
			rawMessage,
			fromPlayer.UserId,
			Enum.TextFilterContext.PublicChat
		)
		if toPlayer then
			return filterResult:GetChatForUserAsync(toPlayer.UserId)
		else
			return filterResult:GetNonChatStringForBroadcastAsync()
		end
	end)
	if ok and type(result) == "string" and result ~= "" then
		return result
	else
		return rawMessage
	end
end

local function getFinalMessage(fromPlayer, rawMessage, toPlayer)
	if containsBlockedAnatomyAnywhere(rawMessage) then
		return BLOCK_TEXT, true, "sexual_content"
	end

	local filtered = robloxFilter(fromPlayer, rawMessage, toPlayer)

	if not filtered or filtered == "" or type(filtered) ~= "string" then
		filtered = rawMessage
	end

	filtered = filtered:match("^%s*(.-)%s*$") or filtered
	return filtered, false, ""
end

end

OwnerTagEvent.OnServerEvent:Connect(function(player, action, payload)
	if player.UserId ~= OWNER_USER_ID then
		return
	end

	payload = type(payload) == "table" and payload or {}

	if action == "request" then
		sendOwnerTagSync(player)
		return
	end

	if action == "setTag" then
		local userId = resolveUserId(payload.target or payload.userId or payload.username)

		if not userId then
			sendSystemMessage(player, "Could not find that player/user.")
			return
		end

		local role = makeRole(payload.tag, payload.color)
		if not role then
			sendSystemMessage(player, "Tag cannot be empty.")
			return
		end

		ownerTagData.users[tostring(userId)] = role

		if ownerTagData.disabledDefaults then
			ownerTagData.disabledDefaults[tostring(userId)] = nil
		end

		saveOwnerTags()
		broadcastPlayerTag(userId)
		sendOwnerTagSync(player)

		sendSystemMessage(player, "Tag saved for UserId " .. tostring(userId) .. ".")
		return
	end

	if action == "removeTag" then
		local userId = resolveUserId(payload.target or payload.userId or payload.username)

		if not userId then
			sendSystemMessage(player, "Could not find that player/user.")
			return
		end

		ownerTagData.users[tostring(userId)] = nil

		if payload.suppressDefault == true then
			if not ownerTagData.disabledDefaults then
				ownerTagData.disabledDefaults = {}
			end

			ownerTagData.disabledDefaults[tostring(userId)] = true
		else
			if ownerTagData.disabledDefaults then
				ownerTagData.disabledDefaults[tostring(userId)] = nil
			end
		end

		saveOwnerTags()
		broadcastPlayerTag(userId)
		sendOwnerTagSync(player)

		sendSystemMessage(player, "Custom tag removed for UserId " .. tostring(userId) .. ".")
		return
	end

	if action == "ownerMessage" then
		local msg = tostring(payload.message or "")
		msg = msg:match("^%s*(.-)%s*$") or ""

		if msg == "" then return end
		if #msg > MAX_MESSAGE_LENGTH then
			msg = msg:sub(1, MAX_MESSAGE_LENGTH)
		end

		for _, plr in ipairs(Players:GetPlayers()) do
			local finalText = robloxFilter(player, msg, plr)
			ChatEvent:FireClient(
				plr,
				player.Name,
				finalText,
				nil,
				false,
				player,
				getTagForUserId(player.UserId)
			)
		end

		return
	end
end)

ChatEvent.OnServerEvent:Connect(function(player, message, whisperTarget)
	local blocked, reason = isToxic(message)

	if blocked then
		logModeration(player, message, reason or "harassment", "BLOCKED")
		sendSystemMessage(
			player,
			"Your message was adjusted to keep chat safe (" .. (reason or "unsafe content") .. ")."
		)
		return
	end

	if typeof(message) ~= "string" then 
		return 
	end

	if message:match("^rbxassetid://") or message:match("^rbxsatestid://") then
		if lastStickerTime[player] == nil then
			lastStickerTime[player] = 0
		end
		if not stickerCooldowns then
			stickerCooldowns = {}
		end
		local now = os.time()
		local timeSinceLastSticker = now - lastStickerTime[player]
		if timeSinceLastSticker < STICKER_COOLDOWN then
			local timeLeft = STICKER_COOLDOWN - timeSinceLastSticker
			stickerCooldowns[player] = now + STICKER_COOLDOWN
			sendSystemMessage(player, "⏳ Please wait " .. math.ceil(timeLeft) .. " seconds before sending another sticker")
			return
		end
		lastStickerTime[player] = now
		stickerCooldowns[player] = now + STICKER_COOLDOWN
		local StickerCooldownEvent = ReplicatedStorage:FindFirstChild("StickerCooldownEvent")
		if not StickerCooldownEvent then
			StickerCooldownEvent = Instance.new("RemoteEvent")
			StickerCooldownEvent.Name = "StickerCooldownEvent"
			StickerCooldownEvent.Parent = ReplicatedStorage
		end
		StickerCooldownEvent:FireClient(player, lastStickerTime[player] + STICKER_COOLDOWN)
		for _, plr in ipairs(Players:GetPlayers()) do
			ChatEvent:FireClient(plr, "Sticker", message, nil, false, player)
		end
		return
	end

	if matchesHardBlock(message) then
		logModeration(player, message, "explicit_threat", "HARD_BLOCK_IMAGE_PHRASE")
		sendSystemMessage(
			player,
			"Your message was blocked due to dangerous or threatening content."
		)
		return
	end

	do
		local joined = collapseRuns(string.lower(message):gsub("[^a-z]", ""))
		if joined:find("hitlerwasright", 1, true) then
			logModeration(player, message, "extremism", "HARD_EARLY_BLOCK")
			sendSystemMessage(player, "Your message was blocked due to harmful or extremist content.")
			return
		end
	end

	do
		local joined = collapseRuns(string.lower(message):gsub("[^a-z]", ""))
		local fuzzyTargets = {
			"diddy", "pdiddy", "seancombs",
			"epstein", "ebstein", "ebstean", "epstean",
		}
		for _, t in ipairs(fuzzyTargets) do
			if joined:find(t, 1, true) then
				logModeration(player, message, "extreme_figure", "HARD_EARLY_BLOCK")
				sendSystemMessage(player, "Your message was blocked due to inappropriate content.")
				return
			end
		end
	end

	if typeof(whisperTarget) == "Instance" and whisperTarget:IsA("Player") then
		local targetPlayer = whisperTarget
		if targetPlayer == player then
			sendSystemMessage(player, "You can't whisper to yourself")
			return
		end
		local shouldBlockMessage, blockReason = shouldBlock(player, message)
		if shouldBlockMessage then
			logModeration(player, message, blockReason, "BLOCKED_WHISPER")
			sendSystemMessage(player, "Your whisper was blocked: " .. tostring(blockReason))
			return
		end
		local filteredForSender, _, _ = getFinalMessage(player, message, player)
		local filteredForTarget, _, _ = getFinalMessage(player, message, targetPlayer)
		ChatEvent:FireClient(player, "You -> " .. targetPlayer.Name, filteredForSender, nil, true, player, getTagForUserId(player.UserId))
		ChatEvent:FireClient(targetPlayer, player.Name, filteredForTarget, nil, true, player, getTagForUserId(player.UserId))
		logModeration(player, "Whisper to " .. targetPlayer.Name .. ": " .. message, "whisper", "SENT")
		return
	end

	local isTeamChat = (whisperTarget == "team")
	if isTeamChat and not player.Team then
		sendSystemMessage(player, "You need to be on a team to use team chat! Switching to All Chat.")
		isTeamChat = false
		whisperTarget = "all"
	end

	if typeof(message) ~= "string" then 
		return 
	end

	if matchesHardBlock(message) then
		logModeration(player, message, "explicit_threat", "HARD_BLOCK_IMAGE_PHRASE")
		sendSystemMessage(
			player,
			"Your message was blocked due to dangerous or threatening content."
		)
		return
	end

	do
		local joined = collapseRuns(string.lower(message):gsub("[^a-z]", ""))
		if joined:find("hitlerwasright", 1, true) then
			logModeration(player, message, "extremism", "HARD_EARLY_BLOCK")
			sendSystemMessage(player, "Your message was blocked due to harmful or extremist content.")
			return
		end
	end

	local isTeamChat2 = (whisperTarget == "team")
	if isTeamChat2 and not player.Team then
		sendSystemMessage(player, "You need to be on a team to use team chat! Switching to All Chat.")
		isTeamChat = false
		whisperTarget = "all"
	end

	if #message > MAX_EXTREME_LENGTH then
		sendSystemMessage(player, "Message too long. Maximum length is " .. MAX_EXTREME_LENGTH .. " characters.")
		return
	end

	message = string.sub(message, 1, MAX_MESSAGE_LENGTH)

	local trimmedMessage = message:match("^%s*(.-)%s*$") or message
	if trimmedMessage == "" then 
		return 
	end
	message = trimmedMessage

	if handleEmoteCommand(player, message) then
		return
	end

	if lastMessageTime[player] == nil then
		lastMessageTime[player] = {}
		messageHistory[player] = {}
		blockedHistory[player] = {}
		muteUntil[player] = 0
	end

	if not spamState[player.UserId] then
		spamState[player.UserId] = { count = 0, windowStart = os.time() }
	end

	local now = os.time()
	local spamInfo = spamState[player.UserId]

	if now - spamInfo.windowStart > SPAM_WINDOW_SECONDS then
		spamInfo.count = 0
		spamInfo.windowStart = now
	end

	local shouldBlockMessage, blockReasonCheck = shouldBlock(player, message)

	if shouldBlockMessage then
		if shouldBlockMessage and
			(blockReasonCheck == "contact_platform"
				or blockReasonCheck == "contact_phrase"
				or blockReasonCheck == "contact_url") then

			logModeration(player, message, blockReasonCheck, "BLOCKED_OFF_PLATFORM")
			sendSystemMessage(
				player,
				"Off-platform contact is not allowed in this chat."
			)
			messageHistory[player] = {}
			blockedHistory[player] = {}
			lastMessageTime[player] = {}
			spamState[player.UserId] = { count = 0, windowStart = os.time(), lastText = "" }
			return
		end

		logModeration(player, message, blockReasonCheck, "BLOCKED")

		if blockReasonCheck == "extremism" or blockReasonCheck == "hate_violence" then
			sendSystemMessage(
				player,
				"Your message was blocked due to harmful or extremist content."
			)
			return
		end

		local displayReason
		if blockReasonCheck == "scam" then
			displayReason = "scam or fake reward"
		elseif blockReasonCheck == "cringe" then
			displayReason = "message not appropriate for this chat"
		elseif blockReasonCheck == "profanity" then
			displayReason = "inappropriate language"
		elseif blockReasonCheck == "single_letter"
			or blockReasonCheck == "gibberish_letters"
			or blockReasonCheck == "alnum_random"
			or blockReasonCheck == "number_only"
			or blockReasonCheck == "toxic_spam"
			or blockReasonCheck == "spam_repeat"
			or blockReasonCheck == "spam_rate" then
			displayReason = "please don't spam random letters or numbers"
		elseif blockReasonCheck == "non_english" then
			displayReason = "English only chat"
		elseif blockReasonCheck == "pii_real_name" then
			displayReason = "personal information (real name)"
		elseif blockReasonCheck:find("pii_") then
			displayReason = "personal information"
		elseif blockReasonCheck:find("age_") then
			displayReason = "age discussion"
		elseif blockReasonCheck == "contact_platform" then
			displayReason = "off-platform contact"
		elseif blockReasonCheck == "dating" then
			displayReason = "dating requests"
		elseif blockReasonCheck == "excessive_repeats" then
			displayReason = "suspicious formatting"
		elseif blockReasonCheck == "contact_image_url" then
			displayReason = "image sharing"
		else
			displayReason = "unsafe content"
		end

		sendSystemMessage(
			player,
			"Your message was adjusted to keep chat safe (" .. displayReason .. ")."
		)

		if player.UserId ~= OWNER_USER_ID then
			blockedHistory[player] = blockedHistory[player] or {}
			if blockReasonCheck ~= "contact_platform"
				and blockReasonCheck ~= "contact_phrase"
				and blockReasonCheck ~= "contact_url" then
				table.insert(blockedHistory[player], blockReasonCheck)
			end

			if #blockedHistory[player] > MUTE_THRESHOLD * 2 then
				table.remove(blockedHistory[player], 1)
			end

			if #blockedHistory[player] >= MUTE_THRESHOLD then
				for _, r in ipairs(blockedHistory[player]) do
					if r == "contact_platform" or r == "contact_phrase" or r == "contact_url" then
						blockedHistory[player] = {}
						return
					end
				end
				muteUntil[player] = os.time() + MUTE_DURATION
				blockedHistory[player] = {}
				sendSystemMessage(player, MUTE_TEXT)
			end
		end

		spamState[player.UserId].lastText = ""
		if blockReasonCheck == "contact_platform" or blockReasonCheck == "contact_phrase" or blockReasonCheck == "contact_url" then
			messageHistory[player] = {}
			lastMessageTime[player] = {}
		end
		blockedHistory[player] = {}
		spamState[player.UserId].lastText = ""
		return
	end

	table.insert(lastMessageTime[player], now)
	table.insert(messageHistory[player], message)

	if #lastMessageTime[player] > SPAM_MAX_IN_WINDOW then
		table.remove(lastMessageTime[player], 1)
		table.remove(messageHistory[player], 1)
	end

	local cleanSpamText = normalizeSpamMessage(message)
	local spamInfo2 = spamState[player.UserId]

	if cleanSpamText ~= "" then
		if spamInfo2.lastText == cleanSpamText then
			spamInfo2.repeatCount = (spamInfo2.repeatCount or 1) + 1
		else
			spamInfo2.lastText = cleanSpamText
			spamInfo2.repeatCount = 1
		end

		if spamInfo2.repeatCount > REPEAT_MESSAGE_LIMIT then
			sendSystemMessage(player, WARN_SPAM_TEXT)
			logModeration(player, message, "spam_repeat", "BLOCKED")

			table.remove(lastMessageTime[player])
			table.remove(messageHistory[player])

			if spamInfo2.repeatCount >= REPEAT_MESSAGE_MUTE_LIMIT then
				muteUntil[player] = os.time() + MUTE_DURATION
				spamInfo2.repeatCount = 0
				spamInfo2.lastText = ""
				sendSystemMessage(player, MUTE_TEXT)
			end

			return
		end
	end

	local timeSinceLast = now - (lastMessageTime[player][#lastMessageTime[player] - 1] or 0)
	if timeSinceLast < SPAM_COOLDOWN and #lastMessageTime[player] > 1 then
		sendSystemMessage(player, WARN_SPAM_TEXT)
		logModeration(player, message, "spam_rate", "WARNED")
		table.remove(lastMessageTime[player])
		table.remove(messageHistory[player])
		return
	end

	if type(muteUntil[player]) == "number" then
		if muteUntil[player] > now then
			sendSystemMessage(player, MUTE_TEXT)
			table.remove(lastMessageTime[player])
			table.remove(messageHistory[player])
			return
		else
			muteUntil[player] = 0
		end
	end

	if containsBlockedAnatomyAnywhere(message) then
		logModeration(player, message, "sexual_content", "HARD_SEND_BLOCK")
		return
	end

	pcall(function()
		channel:SendAsync(message)
	end)

	local playerMessages = {}

	for _, plr in ipairs(Players:GetPlayers()) do
		local finalText, blocked, reason = getFinalMessage(player, message, plr)

		if blocked then
			finalText = BLOCK_TEXT
		end

		if not finalText or finalText == "" or type(finalText) ~= "string" then
			finalText = message
		end

		local trimmed = finalText:match("^%s*(.-)%s*$")
		if trimmed and trimmed ~= "" then
			finalText = trimmed
		else
			finalText = message
		end

		finalText = tostring(finalText)

		if finalText == "" then
			finalText = message
		end

		playerMessages[plr] = { text = finalText, blocked = blocked, reason = reason }
	end

	local senderTeam = player.Team

	for _, plr in ipairs(Players:GetPlayers()) do
		local shouldSend = true

		if serverMutes[plr.UserId] and serverMutes[plr.UserId][player.UserId] then
			shouldSend = false
		end

		if isTeamChat then
			if not senderTeam then
				shouldSend = false
			elseif plr.Team ~= senderTeam then
				shouldSend = false
			end
		end

		if shouldSend then
			local msgData = playerMessages[plr]
			if msgData and msgData.text and msgData.text ~= "" then
				local textToSend = tostring(msgData.text)
				textToSend = textToSend:match("^%s*(.-)%s*$") or textToSend
				if textToSend and textToSend ~= "" then
					ChatEvent:FireClient(plr, player.Name, textToSend, nil, false, player, getTagForUserId(player.UserId))
				end
			end
		end
	end

	for _, plr in ipairs(Players:GetPlayers()) do
		local shouldSend = true

		if serverMutes[plr.UserId] and serverMutes[plr.UserId][player.UserId] then
			shouldSend = false
		end

		if isTeamChat then
			if not senderTeam then
				shouldSend = false
			elseif plr.Team ~= senderTeam then
				shouldSend = false
			end
		end

		if shouldSend then
			local msgData = playerMessages[plr]
			if msgData and msgData.text and msgData.text ~= "" then
				local textToSend = tostring(msgData.text)
				textToSend = textToSend:match("^%s*(.-)%s*$") or textToSend
				if textToSend and textToSend ~= "" then
					ChatEvent:FireClient(plr, "BUBBLE", textToSend, nil, false, player)
				end
			end
		end
	end
end)

Players.PlayerRemoving:Connect(function(plr)
	lastMessageTime[plr] = nil
	messageHistory[plr] = nil
	blockedHistory[plr] = nil
	muteUntil[plr] = nil
	spamState[plr.UserId] = nil
	lastStickerTime[plr] = nil
	stickerCooldowns[plr] = nil
	if typingState[plr] then
		if typingState[plr].thread then task.cancel(typingState[plr].thread) end
		typingState[plr] = nil
	end
end)

		channel:AddUserAsync(plr.UserId)
	end)

	task.wait(1)
	sendSystemMessage(plr, "Chat loaded. All messages go through safety filtering. English-only messages.")

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= plr then
			sendSystemMessage(player, plr.Name .. " has joined the game")
		end
	end

	restoreMuteStateForPlayer(plr)
	notifyExistingPlayersOfRejoin(plr)

	broadcastPlayerTag(plr.UserId)

	if plr.UserId == OWNER_USER_ID then
		sendOwnerTagSync(plr)
	end
end)
